namespace OtoServisApi.Controllers
{
    [ApiController]
    [Route("api/[controller]")]
    public class AdminController : ControllerBase
    {
        private readonly IAuthService _authService;

        public AdminController(IAuthService authService)
        {
            _authService = authService;
        }

        // --- SADECE SUPERADMIN KULLANABİLİR ---
        [HttpPost("admin-ekle")]
        [Authorize(Roles = "SuperAdmin")]
        public async Task<IActionResult> AdminEkle([FromBody] PersonelEkleDto dto)
        {
            if (!ModelState.IsValid) return BadRequest(ModelState);

            // Güvenlik: DTO içinden gelen Rol verisini görmezden geliyoruz. 
            // SuperAdmin bu ucu kullandığı an eklenecek kişi zorunlu "Admin" olur.
            var sonuc = await _authService.PersonelEkleAsync(dto, "Admin");

            if (!sonuc.BasariliMi) return BadRequest(new { Mesaj = sonuc.Mesaj });

            return Ok(new { Mesaj = "Yeni Admin başarıyla eklendi." });
        }

        // --- ADMIN VE SUPERADMIN KULLANABİLİR ---
        [HttpPost("personel-ekle")]
        [Authorize(Roles = "SuperAdmin,Admin")]
        public async Task<IActionResult> PersonelEkle([FromBody] PersonelEkleDto dto)
        {
            if (!ModelState.IsValid) return BadRequest(ModelState);

            // Güvenlik Kalkanı: Admin, kendi kendine SuperAdmin veya başka bir Admin ekleyemez!
            var yasakliRoller = new List<string> { "SuperAdmin", "Admin" };

            if (yasakliRoller.Any(r => r.Equals(dto.Rol, StringComparison.OrdinalIgnoreCase))) // büyük küçük harf duyarlılığını ortadan kaldırıyoruzz
            if (yasakliRoller.Contains(dto.Rol))
            {
                return BadRequest(new { Mesaj = "Bu yetki seviyesinde bir personel ekleme izniniz yok." });
            }

            // Güvenlikten geçtiyse, kullanıcının belirlediği rolü ("Usta", "Vezne" vb.) servise yolla
            var sonuc = await _authService.PersonelEkleAsync(dto, dto.Rol);
            

            if (!sonuc.BasariliMi) return BadRequest(new { Mesaj = sonuc.Mesaj });

            return Ok(new { Mesaj = $"{dto.Rol} rolündeki personel başarıyla eklendi." });
        }
    }
}

    {
        private readonly IAuthService _authService;

        public AuthController(IAuthService authService, IKullaniciService kullaniciService)
        {

            _kullaniciService = kullaniciService;
        }

        [AllowAnonymous]
        [HttpPost("register")]
        public async Task<IActionResult> Register([FromBody] RegisterDto dto)
        {
            try
            {
                var sonuc = await _authService.RegisterAsync(dto);
                return Ok(new { Mesaj = sonuc });
            }
            catch (Exception ex)
            {
                return BadRequest(new { Hata = ex.Message });
            }
        }

        [AllowAnonymous]
        [HttpPost("login")]
        public async Task<IActionResult> Login([FromBody] LoginDto dto)
        {
            try
            {
                var token = await _authService.LoginAsync(dto);
                return Ok(new { Token = token });
            }
            catch (Exception ex)
            {
                return BadRequest(new { Hata = ex.Message });
            }
        }

        [Authorize]
        [HttpPost("logout")]
        public async Task<IActionResult> Logout([FromServices] IDistributedCache cache)
        {
            // 1. İstekten gelen ham Bearer token'ı al
            var rawToken = Request.Headers["Authorization"].ToString().Replace("Bearer ", "").Trim();
            if (string.IsNullOrEmpty(rawToken))
                return BadRequest("Token bulunamadı.");

            // 2. Token'ın süresinin bitmesine kaç dakika/saat kaldığını hesapla
            var handler = new JwtSecurityTokenHandler();
            var jwtToken = handler.ReadJwtToken(rawToken);
            var kalanSure = jwtToken.ValidTo - DateTime.UtcNow;

            // Eğer süre zaten dolmuşsa işlem yapmaya gerek yok
            if (kalanSure <= TimeSpan.Zero)
                return Ok(new { Mesaj = "Oturum zaten sonlanmış." });

            // 3. Token'ı kalan ömrü kadar Redis'e 'kara liste' olarak kaydet
            var cacheKey = $"blacklist:{rawToken}";
            var cacheOptions = new DistributedCacheEntryOptions
            {
                AbsoluteExpirationRelativeToNow = kalanSure
            };

            await cache.SetStringAsync(cacheKey, "revoked", cacheOptions);

            return Ok(new { Mesaj = "Başarıyla çıkış yapıldı. Token geçersiz kılındı." });
        }

        [AllowAnonymous]
        [HttpPost("sifremi-unuttum")]
        public async Task<IActionResult> SifremiUnuttum([FromBody] SifremiUnuttumDto dto)

        public async Task<IActionResult> SifreGuncelle([FromBody] SifreGuncelleDto dto)

﻿using Microsoft.AspNetCore.Authorization;
using Microsoft.AspNetCore.Mvc;
using Microsoft.Extensions.Caching.Distributed;
using OtoServis.Data.Models;
using OtoServis.Data.Repositories;
using System.Text.Json;

namespace OtoServisApi.Controllers
{
    [Route("api/[controller]")]
    [ApiController]
    public class HizmetTipleriController : ControllerBase
    {
        private readonly IRepository<HizmetTipi> _hizmetTipiRepository;
        private readonly IDistributedCache _cache;

        public HizmetTipleriController(IRepository<HizmetTipi> hizmetTipiRepository, IDistributedCache cache)
         
        {
            _hizmetTipiRepository = hizmetTipiRepository;
            _cache = cache;
        }

        [HttpGet]
        [AllowAnonymous]
        public async Task<ActionResult<IEnumerable<HizmetTipi>>> GetHizmetTipleri()
        {
            string cacheKey = "tum_hizmetler";

            // 1. ADIM: Redis'e bağlanmayı DENE
            try
            {
                var cachedData = await _cache.GetStringAsync(cacheKey);
                if (!string.IsNullOrEmpty(cachedData))
                {
                    var jsonHizmetler = JsonSerializer.Deserialize<List<HizmetTipi>>(cachedData);
                    return Ok(jsonHizmetler);
                }
            }
            catch (Exception)
            {
                // Eğer Redis sunucusu kapalıysa veya çöktüyse kod buraya düşer.
                // Hata fırlatmayı engelliyoruz ki sistem çökmesin, doğrudan SQL'e geçsin!
               
            }

            // 2. ADIM: Redis'te yoksa VEYA Redis kapalıysa SQL Server'dan getir
            var hizmetler = (await _hizmetTipiRepository.GetAllAsync()).ToList();

            // 3. ADIM: Yeni veriyi Redis'e yazmayı DENE (Yine hata fırlatmasını engelliyoruz)
            try
            {
                var cacheOptions = new DistributedCacheEntryOptions
                {
                    AbsoluteExpirationRelativeToNow = TimeSpan.FromMinutes(30)
                };
                await _cache.SetStringAsync(cacheKey, JsonSerializer.Serialize(hizmetler), cacheOptions);
            }
            catch (Exception)
            {
                // Redis kapalıysa yazamaz, sorun değil. Kullanıcıya verisini dönmeye devam et.
            }

            return Ok(hizmetler);
        }

        [HttpPost]
        [Authorize(Roles = "SuperAdmin,Admin")]
        public async Task<ActionResult<HizmetTipi>> PostHizmetTipi(HizmetTipi hizmetTipi, [FromServices] IDistributedCache cache)
        {
            await _hizmetTipiRepository.AddAsync(hizmetTipi);
            await _hizmetTipiRepository.SaveChangesAsync();

            // ÖNBELLEĞİ BOŞALT: Yeni hizmet eklendiği için eski listeyi çöpe atıyoruz
            await cache.RemoveAsync("tum_hizmetler");

            return CreatedAtAction(nameof(GetHizmetTipleri), new { id = hizmetTipi.Id }, hizmetTipi);
        }
    }

﻿using Microsoft.AspNetCore.Authorization;
using Microsoft.AspNetCore.Mvc;
using OtoServisApi.DTOs;
using OtoServis.Data.Models;
using OtoServis.Data.Repositories;
using System.Security.Claims;

namespace OtoServisApi.Controllers
{
    [Route("api/[controller]")]
    [ApiController]
    [Authorize] // Token'ı olan (müşteri) ödeme yapabilir
    public class OdemeController : ControllerBase
    {
        private readonly IRepository<Fatura> _faturaRepository;
        private readonly IRepository<Randevu> _randevuRepository;


        public OdemeController(IRepository<Fatura> faturaRepository, IRepository<Randevu> randevuRepository)
        {
            _faturaRepository = faturaRepository;
            _randevuRepository = randevuRepository;
        }


        [Authorize(Roles = "Veznedar,SuperAdmin")]
        [HttpPost("randevu/{randevuId}/fiziksel-ode")]
        public async Task<IActionResult> FizikselOdemeAlByRandevu(int randevuId, [FromQuery] string odemeTuru)
        {
            // 1. Önce faturanın var olup olmadığını kontrol ediyoruz
            var fatura = await _faturaRepository.GetAsync(f => f.RandevuId == randevuId);
            if (fatura == null)
            {
                return NotFound($"#{randevuId} numaralı randevuya ait kesilmiş bir fatura sistemde bulunamadı.");
            }

            // 2. Fatura var ama daha önce ödenmiş mi?
            if (fatura.OdendiMi)
            {
                return BadRequest("Bu fatura zaten daha önce ödenmiş ve kapatılmıştır.");
            }

            fatura.OdendiMi = true;
            fatura.OdemeTarihi = DateTime.Now;
            fatura.OdemeYontemi = odemeTuru;
            fatura.TahsilEdenKullaniciId = int.Parse(User.FindFirst(ClaimTypes.NameIdentifier)!.Value);

            await _faturaRepository.UpdateAsync(fatura);
            await _faturaRepository.SaveChangesAsync();

            return Ok(new
            {
                Mesaj = $"#{randevuId} nolu randevunun ödemesi veznede ({odemeTuru}) olarak tahsil edildi.",
                FaturaId = fatura.Id,
                RandevuId = randevuId
            });
        }

        [Authorize(Roles = "Veznedar,SuperAdmin")]
        [HttpPut("{id}/teslim-et")]
        public async Task<IActionResult> AraciTeslimEt(int id)
        {
            var randevu = await _randevuRepository.GetAsync(r => r.Id == id);
            if (randevu == null) return NotFound("Randevu bulunamadı.");

            if (randevu.Durum != RandevuDurumu.Tamamlandi)
            {
                return BadRequest("Usta henüz bakım/onarım işlemlerini tamamlamadı.");
            }

            var fatura = await _faturaRepository.GetAsync(f => f.RandevuId == id);
            if (fatura == null || !fatura.OdendiMi)
            {
                return BadRequest("Ödeme onayı bulunamadı! Fatura tahsil edilmeden araç teslim edilemez.");
            }

            randevu.TeslimTarihi = DateTime.Now;
            randevu.TeslimEdenKullaniciId = int.Parse(User.FindFirst(ClaimTypes.NameIdentifier)!.Value);
            randevu.Durum = RandevuDurumu.Tamamlandi; // veya RandevuDurumu.TeslimEdildi
            await _randevuRepository.SaveChangesAsync();
            return Ok(new
            {

                Mesaj = "Ödeme kontrolü tamamlandı. Araç ve anahtar müşteriye teslim edilebilir.",
                RandevuId = id
            });
        }
    }

    {
        private readonly IRepository<YedekParca> _yedekParcaRepository;
        private readonly IDistributedCache _cache;

        public YedekParcalarController(IRepository<YedekParca> yedekParcaRepository, IDistributedCache cache)
        {
            _yedekParcaRepository = yedekParcaRepository;
            _cache = cache;
        }

        [Authorize(Roles = "SuperAdmin, Admin, DepoSorumlusu, Usta")]
        [HttpGet("tum-stoklar")]
        public async Task<IActionResult> GetYedekParcalar()
        {
            string cacheKey = "tum_yedek_parcalar";

            // 1. Önce Redis'e bak (Cache Hit)
            var cachedData = await _cache.GetStringAsync(cacheKey);
            if (!string.IsNullOrEmpty(cachedData))
            {
                var jsonParcalar = JsonSerializer.Deserialize<List<YedekParca>>(cachedData);
                return Ok(jsonParcalar);
            }

            // 2. Redis'te yoksa veritabanından çek (Cache Miss)
            var parcalar = (await _yedekParcaRepository.GetAllAsync()).ToList();

            // 3. Veriyi 60 dakika geçerli olacak şekilde Redis'e yaz
            var cacheOptions = new DistributedCacheEntryOptions
            {
                AbsoluteExpirationRelativeToNow = TimeSpan.FromMinutes(60)
            };
            await _cache.SetStringAsync(cacheKey, JsonSerializer.Serialize(parcalar), cacheOptions);

            return Ok(parcalar);
        }

        [HttpGet("kritik-stoklar")]

﻿using Microsoft.AspNetCore.Http;
using OtoServis.Data; // DbContext'inin bulunduğu namespace 
using OtoServis.Data.Models;
using System;
using System.Security.Claims;
using System.Threading.Tasks;


namespace OtoServisApi.Middlewares
{
    public class GlobalAuditMiddleware
    {
        private readonly RequestDelegate _next;

        public GlobalAuditMiddleware(RequestDelegate next)
        {
            _next = next;
        }

        // DbContext'i constructor'da değil, InvokeAsync içinde alıyoruz çünkü DbContext "Scoped" (istek başına) çalışır.
        
        public async Task InvokeAsync(HttpContext context, ServisDbContext dbContext)
        {
            // 1. İsteği API'nin derinliklerine (Controller'a) gönder ve cevabı (Red veya Onay) bekle
            await _next(context);

            // 2. Sadece /api/ ile başlayan gerçek istekleri logla (Swagger arayüz dosyalarını es geçmek için)
            if (context.Request.Path.StartsWithSegments("/api"))
            {
                var statusCode = context.Response.StatusCode;
                var metod = context.Request.Method;
                var yol = context.Request.Path;

                // Sadece GET (Okuma) isteklerini veya Hatalı/Yetkisiz (401, 403, 500 vb.) durumları logla 
                //Başarısız POST/PUT/DELETE (Durum Kodu 400, 401, 403, 500) durum kodları bunlar ve bu eylemler de Global Middleware tarafından yakalanıp loglanır
                // (POST, PUT, DELETE zaten başarılı olursa 1. kalkanımız olan DbContext tarafından detaylıca loglanıyor!)
                if (metod == "GET" || statusCode >= 400)
                {
                    // Kullanıcı veya IP tespiti
                    string aktifKullaniciId = "Ziyaretci IP: " + (context.Connection?.RemoteIpAddress?.ToString() ?? "Bilinmiyor");

                    if (context.User?.Identity?.IsAuthenticated == true)
                    {
                        aktifKullaniciId = context.User.FindFirst(ClaimTypes.NameIdentifier)?.Value
                                        ?? context.User.FindFirst("id")?.Value
                                        ?? "Bilinmeyen Kullanici";
                    }

                    // Durum koduna göre insancıl bir işlem tipi belirle
                    string islemTipi = statusCode switch
                    {
                        401 => "YETKİSİZ ERİŞİM (401)",
                        403 => "YASAKLI ERİŞİM (403)",
                        >= 500 => "SUNUCU HATASI (500)",
                        _ => "SİSTEM OKUMA (GET)"
                    };

                    // Log nesnesini oluştur
                    var yeniLog = new SistemLog
                    {
                        IslemTipi = islemTipi,
                        TabloAdi = "API_ENDPOINT",
                        Detay = $"{metod} isteği yapıldı. Hedef: {yol} | Sonuç Kodu: {statusCode}",
                        Tarih = DateTime.UtcNow.AddHours(3),
                        KullaniciId = aktifKullaniciId
                    };

                    dbContext.SistemLoglari.Add(yeniLog);
                    await dbContext.SaveChangesAsync();
                }
            }
        }
    }
}

    public class AuthService : IAuthService
    {
        private readonly IRepository<Kullanici> _kullaniciRepository;
        private readonly IConfiguration _configuration;
        
        private readonly IPublishEndpoint _publishEndpoint; // Mail servisi yerine RabbitMQ geldi
        public AuthService(
            IRepository<Kullanici> kullaniciRepository,
            IConfiguration configuration,
           
            IPublishEndpoint publishEndpoint
            )
        {
            _kullaniciRepository = kullaniciRepository;
            _configuration = configuration;
            
            _publishEndpoint = publishEndpoint;
        }

        public async Task<string> RegisterAsync(RegisterDto dto)
        {
            // 1. Bu e-postayla daha önce bir kayıt var mı? (Misafir de olabilir, müşteri de)
            var mevcutKullanici = await _kullaniciRepository.GetAsync(k => k.Email == dto.Email);

            if (mevcutKullanici != null)
            {
                // EĞER DAHA ÖNCE MİSAFİR OLARAK GELMİŞSE: Yeni satır açma, üstüne yaz!
                if (mevcutKullanici.Rol == "Misafir" || mevcutKullanici.Rol == "misafir" || string.IsNullOrEmpty(mevcutKullanici.SifreHash))
                {
                    mevcutKullanici.AdSoyad = dto.AdSoyad; 
                    mevcutKullanici.SifreHash = BCrypt.Net.BCrypt.HashPassword(dto.Sifre);
                    mevcutKullanici.Rol = "Musteri";

                    await _kullaniciRepository.UpdateAsync(mevcutKullanici);
                    await _kullaniciRepository.SaveChangesAsync();
                    return "Misafir kaydınız kalıcı müşteri hesabına dönüştürüldü.";
                }

                throw new Exception("Bu e-posta adresi ile zaten kayıtlı bir hesap bulunuyor.");
            }

            // 2. Hiç yoksa normal yeni kayıt aç
            var sifreHash = BCrypt.Net.BCrypt.HashPassword(dto.Sifre);
            var yeniKullanici = new Kullanici
            {
                AdSoyad = dto.AdSoyad,
                Email = dto.Email,
                SifreHash = sifreHash,
                Rol = dto.Rol
            };

            await _kullaniciRepository.AddAsync(yeniKullanici);
            await _kullaniciRepository.SaveChangesAsync();

            return "Kayıt işlemi başarıyla tamamlandı.";
        }

        public async Task<string> LoginAsync(LoginDto dto)
        {
            // KESİN ÇÖZÜM: E-postası eşleşen VEEEE Şifresi boş OLMAYAN (Gerçek) hesabı getir!
            var kullanici = await _kullaniciRepository.GetAsync(k => k.Email == dto.Email && k.SifreHash != null);

            if (kullanici == null || !BCrypt.Net.BCrypt.Verify(dto.Sifre, kullanici.SifreHash))
                throw new Exception("E-posta adresi veya şifre hatalı.");

            return TokenUret(kullanici);
        }

        public async Task<int> MisafirKaydetVeyaBulAsync(string adSoyad, string email, string telefon)
        {
            // 1. SORGULAMA: Veritabanında bu telefon numarasına VEYA e-postaya sahip biri var mı?
            var mevcutKullanicilar = await _kullaniciRepository.GetAllAsync(
                k => k.Telefon == telefon || k.Email == email);

            var musteri = mevcutKullanicilar.FirstOrDefault();

            // 2. BULUNDU DURUMU (Eski misafir veya zaten kayıtlı bir müşteri)
            if (musteri != null)
            {
                // GÜVENLİK KALKANI: Bulunan kişi Misafir DEĞİLSE (Müşteri, Usta, Admin vs. ise) hata fırlat
                if (musteri.Rol != "Misafir" && musteri.Rol != "misafir")
                {
                    throw new Exception("Bu e-posta adresiyle kayıtlı bir hesabınız bulunuyor. Lütfen önce giriş yapınız.");
                }

                // Eğer rolü cidden misafirse, eski ID'sini ver ki yeni randevu da onun üstüne yazılsın
                return musteri.Id;
            }

            // 3. BULUNAMADI DURUMU (Sisteme tamamen yabancı, ilk defa gelen kişi)
            var yeniMisafir = new Kullanici
            {
                AdSoyad = adSoyad,
                Telefon = telefon,
                Email = email,
                Rol = "Misafir", // Normal müşteri hesabından ayırmak için rolü Misafir yapıyoruz
                SifreHash = null // Şifre null kalıyor çünkü bu kişi sisteme bilinçli bir "Kayıt Ol" işlemi yapmadı
            };

            // 4. KAYIT İŞLEMİ: Yeni misafiri veritabanına ekle
            await _kullaniciRepository.AddAsync(yeniMisafir);
            _kullaniciRepository.SaveChangesAsync();

            // 5. YENİ ID'Yİ DÖNDÜR: Veritabanı tarafından atanan taze ID'yi Controller'a yolluyoruz.
            return yeniMisafir.Id;
        }



                

        public async Task<(bool BasariliMi, string Mesaj)> PersonelEkleAsync(PersonelEkleDto dto, string atanacakRol)
        {
            // 1. E-Posta Tekrarı Kontrolü
            var mevcutKullanici = await _kullaniciRepository.GetAsync(k => k.Email == dto.Email);
            if (mevcutKullanici != null)
            {
                return (false, "Bu e-posta adresiyle kayıtlı bir kullanıcı zaten var.");
            }

            // 2. Şifreyi Hashleme
            string hashliSifre = BCrypt.Net.BCrypt.HashPassword(dto.Sifre);

            // 3. Yeni Entity'yi Oluşturma
            var yeniPersonel = new Kullanici
            {
                AdSoyad = dto.AdSoyad,
                Email = dto.Email,
                Telefon = dto.Telefon,
                SifreHash = hashliSifre,
                Rol = atanacakRol // Controller'dan onaylanıp gelen güvenli rol
            };

            // 4. Veritabanına Kaydetme
            await _kullaniciRepository.AddAsync(yeniPersonel);
            await _kullaniciRepository.SaveChangesAsync();

            return (true, $"{atanacakRol} rolündeki personel başarıyla oluşturuldu.");
        }



        public async Task<(bool BasariliMi, string Mesaj)> SifreSifirlamaKoduGonderAsync(SifremiUnuttumDto dto)
        {
            // KESİN ÇÖZÜM: Şifresi olan gerçek hesabı bul
            var kullanici = await _kullaniciRepository.GetAsync(k => k.Email == dto.Email && k.SifreHash != null );

            if (kullanici == null)
            {
                // Eğer o maille sadece Misafir varsa veya hiç yoksa
                return (false, "Bu e-posta adresiyle kayıtlı şifreli bir hesabınız bulunmuyor. Lütfen önce 'Kayıt Ol' sayfasından hesabınızı oluşturun.");
            }

            // Kırılmaz bir GUID (Token) üretiyoruz
            string resetToken = Guid.NewGuid().ToString();

            kullanici.SifreSifirlamaKodu = resetToken;
            kullanici.SifreSifirlamaKoduBitis = DateTime.Now.AddMinutes(15);

            await _kullaniciRepository.UpdateAsync(kullanici);
            await _kullaniciRepository.SaveChangesAsync();

            string frontendSifirlamaLinki = $"http://localhost:5173/sifre-sifirla?token={Uri.EscapeDataString(resetToken)}&email={Uri.EscapeDataString(kullanici.Email)}";

            string mailIcerik = $"<h2>Şifre Sıfırlama Talebi</h2>" +
                                $"<p>Yeni şifrenizi belirlemek için aşağıdaki bağlantıya tıklayınız:</p>" +
                                $"<a href='{frontendSifirlamaLinki}' style='padding: 10px 20px; background-color: #007bff; color: white; text-decoration: none; border-radius: 5px; display: inline-block;'>Şifremi Sıfırla</a>" +
                                $"<br><br><p>Bu bağlantı 15 dakika sonra geçerliliğini yitirecektir.</p>";

            var mailEvent = new SifreSifirlamaIstendiEvent
            {
                AliciEmail = kullanici.Email,
                Konu = "Oto Servis - Şifre Sıfırlama Bağlantısı",
                HtmlIcerik = mailIcerik
            };

            await _publishEndpoint.Publish(mailEvent);

            return (true, "Şifre sıfırlama bağlantısı e-posta adresinize gönderildi.");
        }




        public async Task<(bool BasariliMi, string Mesaj)> YeniSifreyiKaydetAsync(SifreSifirlaDto dto)
        {
            var kullanici = await _kullaniciRepository.GetAsync(k => k.Email == dto.Email);

            if (kullanici == null || kullanici.SifreSifirlamaKodu != dto.Token)
            {
                return (false, "Geçersiz e-posta adresi veya sıfırlama kodu.");
            }

            if (kullanici.SifreSifirlamaKoduBitis < DateTime.Now)
            {
                return (false, "Sıfırlama kodunun süresi dolmuş. Lütfen tekrar talep ediniz.");
            }

            // Kod doğru ve süresi geçmemiş! Şifreyi BCrypt ile eziyoruz
            kullanici.SifreHash = BCrypt.Net.BCrypt.HashPassword(dto.YeniSifre);

            // Güvenlik için kullanılmış kodu temizliyoruz
            kullanici.SifreSifirlamaKodu = null;
            kullanici.SifreSifirlamaKoduBitis = null;

            await _kullaniciRepository.UpdateAsync(kullanici);
            await _kullaniciRepository.SaveChangesAsync();

            return (true, "Şifreniz başarıyla güncellendi. Yeni şifrenizle giriş yapabilirsiniz.");
        }



        private string TokenUret(Kullanici kullanici)
        {
            var jwtSettings = _configuration.GetSection("JwtSettings");
            var gizliAnahtar = new SymmetricSecurityKey(Encoding.UTF8.GetBytes(jwtSettings["Key"]));
            var imzaKredisi = new SigningCredentials(gizliAnahtar, SecurityAlgorithms.HmacSha256);

            // Token'ın içine gömülecek kimlik bilgileri (Yaka Kartı detayları)
            // AuthService.cs içindeki TokenUret metodu
            var claims = new[]
            {
                // Sadece değişmeyen statik kimlik ve yetki bilgileri
                new Claim(ClaimTypes.NameIdentifier, kullanici.Id.ToString()),
                new Claim(ClaimTypes.Role, kullanici.Rol)
            };

            var token = new JwtSecurityToken(
                issuer: jwtSettings["Issuer"],
                audience: jwtSettings["Audience"],
                claims: claims,
                expires: DateTime.Now.AddHours(3), // Token 3 saat geçerli olsun
                signingCredentials: imzaKredisi
            );

            return new JwtSecurityTokenHandler().WriteToken(token);
        }

            return await _kullaniciRepository.GetAsync(k => k.Id == kullaniciId);

        public async Task<(bool BasariliMi, string Mesaj)> ProfilGuncelleAsync(int kullaniciId, ProfilGuncelleDto dto)
        {
            var kullanici = await _kullaniciRepository.GetAsync(k => k.Id == kullaniciId);
            if (kullanici == null)
                return (false, "Kullanıcı bulunamadı.");

            // E-posta başkası tarafından kullanılıyor mu kontrolü
            var emailKontrol = await _kullaniciRepository.GetAsync(k => k.Email == dto.Email && k.Id != kullaniciId);
            if (emailKontrol != null)
                return (false, "Bu e-posta adresi başka bir kullanıcı tarafından kullanılmaktadır.");

            // Telefon başkası tarafından kullanılıyor mu kontrolü (Eğer telefon girildiyse)
            if (!string.IsNullOrEmpty(dto.Telefon))
            {
                var telKontrol = await _kullaniciRepository.GetAsync(k => k.Telefon == dto.Telefon && k.Id != kullaniciId);
                if (telKontrol != null)
                    return (false, "Bu telefon numarası başka bir kullanıcı tarafından kullanılmaktadır.");
            }

            // Bilgileri güncelle
            kullanici.AdSoyad = dto.AdSoyad;
            kullanici.Email = dto.Email;
            kullanici.Telefon = dto.Telefon;

            await _kullaniciRepository.UpdateAsync(kullanici);
            await _kullaniciRepository.SaveChangesAsync();

            return (true, "Profil bilgileriniz başarıyla güncellendi.");
        }

        public async Task<(bool BasariliMi, string Mesaj)> SifreGuncelleAsync(int kullaniciId, SifreGuncelleDto dto)
        {
            // 1. Kullanıcıyı bul
            var kullanici = await _kullaniciRepository.GetAsync(k => k.Id == kullaniciId);
            if (kullanici == null) return (false, "Kullanıcı bulunamadı.");

            // 2. Mevcut şifreyi doğrula
            if (!BCrypt.Net.BCrypt.Verify(dto.MevcutSifre, kullanici.SifreHash))
            {
                return (false, "Mevcut şifrenizi yanlış girdiniz.");
            }

            // 3. Yeni şifreyi hashle ve kaydet
            kullanici.SifreHash = BCrypt.Net.BCrypt.HashPassword(dto.YeniSifre);
            await _kullaniciRepository.UpdateAsync(kullanici);
            await _kullaniciRepository.SaveChangesAsync();

            return (true, "Şifreniz başarıyla güncellendi.");
        }

﻿using Microsoft.Extensions.DependencyInjection;
using Microsoft.Extensions.Hosting;
using Microsoft.Extensions.Logging;

using OtoServis.Data.Models;
using OtoServis.Data.Repositories;

namespace OtoServisApi.Services
{
    public class KaynakPlanlamaMotoru : BackgroundService
    {
        // Arka plan servisleri sürekli çalıştığı (Singleton) için, 
        // veritabanı gibi anlık (Scoped) servislere ulaşmak adına bir fabrika kullanmalıyız.
        private readonly IServiceScopeFactory _scopeFactory;
        private readonly ILogger<KaynakPlanlamaMotoru> _logger;

        public KaynakPlanlamaMotoru(IServiceScopeFactory scopeFactory, ILogger<KaynakPlanlamaMotoru> logger)
        {
            _scopeFactory = scopeFactory;
            _logger = logger;
        }

        protected override async Task ExecuteAsync(CancellationToken stoppingToken)
        {
            _logger.LogInformation("Kaynak Planlama Motoru çalışmaya başladı...");

            // API ayakta kaldığı sürece bu döngü çalışmaya devam eder
            while (!stoppingToken.IsCancellationRequested)
            {
                try
                {
                    await PlanlamaYapAsync();
                }
                catch (Exception ex)
                {
                    _logger.LogError(ex, "Planlama motorunda bir hata oluştu.");
                }

                // Motor havuzu çok yormamak için her 1 dakikada (60.000 ms) bir çalışır. 
                // Test için bu süreyi 10000 (10 saniye) yapabilirsin.
                await Task.Delay(60000, stoppingToken);
            }
        }

        private async Task PlanlamaYapAsync()
        {
            using var scope = _scopeFactory.CreateScope();
            var randevuRepo = scope.ServiceProvider.GetRequiredService<IRepository<Randevu>>();
            var ustaRepo = scope.ServiceProvider.GetRequiredService<IRepository<Usta>>();
            var istasyonRepo = scope.ServiceProvider.GetRequiredService<IRepository<Istasyon>>();
            var hizmetTipiRepo = scope.ServiceProvider.GetRequiredService<IRepository<HizmetTipi>>();

            var tumRandevular = (await randevuRepo.GetAllAsync()).ToList();

            // Hizmet tiplerinin usta ve lift sürelerini sözlük olarak hafızaya alıyoruz
            var hizmetUstaSureleri = (await hizmetTipiRepo.GetAllAsync())
                .ToDictionary(h => h.Id, h => h.UstaSureDakika > 0 ? h.UstaSureDakika : 30);

            var hizmetLiftSureleri = (await hizmetTipiRepo.GetAllAsync())
                .ToDictionary(h => h.Id, h => h.LiftSureDakika > 0 ? h.LiftSureDakika : 60);

            // 1. BEKLEYENLERİ BUL: "TamamlandiMi" yerine "Durum" kontrolü geldi
            var bekleyenRandevular = tumRandevular
                .Where(r => (r.UstaId == null || r.IstasyonId == null)
                         && r.Durum == RandevuDurumu.Bekliyor
                         && r.Tarih > DateTime.Now)
                .OrderByDescending(r => r.AcilMi ? 1 : 0)
                .ThenBy(r => r.Tarih)
                .ToList();

            
            var ustalar = await ustaRepo.GetAllAsync();
            var istasyonlar = await istasyonRepo.GetAllAsync();

            
            foreach (var randevu in bekleyenRandevular)
            {
                   
                    int ustaSure = hizmetUstaSureleri.ContainsKey(randevu.HizmetTipiId) ? hizmetUstaSureleri[randevu.HizmetTipiId] : 30;
                    int liftSure = hizmetLiftSureleri.ContainsKey(randevu.HizmetTipiId) ? hizmetLiftSureleri[randevu.HizmetTipiId] : 60;

                    DateTime yeniBaslangic = randevu.Tarih;
                    DateTime yeniUstaBitis = randevu.Tarih.AddMinutes(ustaSure);
                    DateTime yeniLiftBitis = randevu.Tarih.AddMinutes(liftSure);

                    // --- USTA İÇİN DİNAMİK ZAMAN KESİŞİM KONTROLÜ ---
                    var doluUstaIdleri = tumRandevular
                        .Where(r => r.UstaId != null && (r.Durum == RandevuDurumu.Bekliyor || r.Durum == RandevuDurumu.Islemde))
                        .Where(r =>
                        {
                            int mevcutUstaSure = hizmetUstaSureleri.ContainsKey(r.HizmetTipiId) ? hizmetUstaSureleri[r.HizmetTipiId] : 30;

                            // DİNAMİZM BURADA: İşleme başlanmışsa gerçek başlama saatini baz al, henüz başlanmamışsa planlananı!
                            DateTime mevcutBaslangic = r.GerceklesenBaslangic ?? r.Tarih;
                            DateTime mevcutUstaBitis = mevcutBaslangic.AddMinutes(mevcutUstaSure);

                            return yeniBaslangic < mevcutUstaBitis && mevcutBaslangic < yeniUstaBitis;
                        })
                        .Select(r => r.UstaId.Value)
                        .ToList();

                    // --- LİFT İÇİN DİNAMİK ZAMAN KESİŞİM KONTROLÜ ---
                    var doluIstasyonIdleri = tumRandevular
                        .Where(r => r.IstasyonId != null && (r.Durum == RandevuDurumu.Bekliyor || r.Durum == RandevuDurumu.Islemde))
                        .Where(r =>
                        {
                            int mevcutLiftSure = hizmetLiftSureleri.ContainsKey(r.HizmetTipiId) ? hizmetLiftSureleri[r.HizmetTipiId] : 60;

                            // DİNAMİZM BURADA: İşleme başlanmışsa gerçek başlama saatini baz al, henüz başlanmamışsa planlananı!
                            DateTime mevcutBaslangic = r.GerceklesenBaslangic ?? r.Tarih;
                            DateTime mevcutLiftBitis = mevcutBaslangic.AddMinutes(mevcutLiftSure);

                            return yeniBaslangic < mevcutLiftBitis && mevcutBaslangic < yeniLiftBitis;
                        })
                        .Select(r => r.IstasyonId.Value)
                        .ToList();

                    //ustanın  uzmanlığına da bakıyoruz:
                    var uygunUsta = ustalar.FirstOrDefault(u =>
                        u.UzmanlikAlani == randevu.HizmetTipi.GerekenUzmanlik &&
                        !doluUstaIdleri.Contains(u.Id)
                    );
                    var uygunIstasyon = istasyonlar.FirstOrDefault(i => !doluIstasyonIdleri.Contains(i.Id) && i.MusaitMi);

                    if (uygunUsta != null && uygunIstasyon != null)
                    {
                        randevu.UstaId = uygunUsta.Id;
                        randevu.IstasyonId = uygunIstasyon.Id;

                        await randevuRepo.UpdateAsync(randevu);
                    await randevuRepo.SaveChangesAsync();

                    _logger.LogInformation($"[DİNAMİK ATAMA] Randevu: {randevu.Id} | Usta Süresi: {ustaSure} dk | Usta: {uygunUsta.AdSoyad} | İstasyon: {uygunIstasyon.Ad}");
                    }
                
            }

            // --- TRAFİK POLİSİ (YÜK DENGELEME - LOAD BALANCING) SİSTEMİ ---

            // 1. Saati geldiği/geçtiği halde HALA bekleyen randevuları bul
            var gecikenRandevular = tumRandevular
                .Where(r => r.UstaId != null
                         && r.Durum == RandevuDurumu.Bekliyor
                         && r.Tarih <= DateTime.Now.AddMinutes(60)) // Saati gelmiş veya max 5 dk kalmış
                .ToList();

            // 2. ŞU AN dükkanda aktif olarak çalışan (Islemde) ustaların ID'lerini bir List<int> olarak topla
            var anlikDoluUstaIdleri = tumRandevular
                .Where(r => r.Durum == RandevuDurumu.Islemde && r.UstaId != null)
                .Select(r => r.UstaId.Value)
                .Distinct()
                .ToList();

        public async Task<bool> SureUzatVeArkadakileriKaydirAsync(int randevuId, SureUzatDto dto)
        {
            var tumRandevular = (await _randevuRepository.GetAllAsync(
                null,
                 r => r.Kullanici,
                 r => r.HizmetTipi,
                 r => r.Arac // Eksik olan Include eklendi
             )).ToList();

            var ustalar = (await _ustaRepository.GetAllAsync()).ToList();

            var randevu = await _randevuRepository.GetByIdAsync(randevuId);
            if (randevu == null) return false;

            if (randevu.Durum == RandevuDurumu.Bekliyor)
            {
                randevu.Durum = RandevuDurumu.Islemde;
                randevu.GerceklesenBaslangic = DateTime.Now;
                await _randevuRepository.UpdateAsync(randevu);
            }

            var arkadakiRandevular = tumRandevular
                .Where(r => r.UstaId == randevu.UstaId && r.Tarih > randevu.Tarih && r.Durum == RandevuDurumu.Bekliyor)
                .OrderBy(r => r.Tarih)
                .ToList();

            TimeSpan mesaiBaslangic = new TimeSpan(9, 0, 0);
            TimeSpan mesaiBitis = new TimeSpan(18, 0, 0);
            TimeSpan molaBaslangic = new TimeSpan(12, 0, 0);
            TimeSpan molaBitis = new TimeSpan(13, 0, 0);

            foreach (var arkaRandevu in arkadakiRandevular)
            {
                int islemSuresi = arkaRandevu.HizmetTipi?.UstaSureDakika ?? 30;
                bool alternatifBulundu = false;
                bool mesaiAsimiOldu = false;

                var maxErtelemeSaati = arkaRandevu.Tarih.AddMinutes(dto.EkSureDakika);
                var orijinalSaat = arkaRandevu.Tarih;
                var denenenSaat = orijinalSaat;

                while (denenenSaat <= maxErtelemeSaati)
                {
                    var denenenBitisSaati = denenenSaat.AddMinutes(islemSuresi);

                    if (denenenSaat.TimeOfDay < molaBitis && denenenBitisSaati.TimeOfDay > molaBaslangic)
                    {
                        denenenSaat = denenenSaat.Date.Add(molaBitis);
                        continue;
                    }

                    if (denenenBitisSaati.TimeOfDay > mesaiBitis)
                    {
                        mesaiAsimiOldu = true;
                        break;
                    }

                    foreach (var usta in ustalar)
                    {
                        if (usta.Id == randevu.UstaId) continue;
                        if (usta.UzmanlikAlani != arkaRandevu.HizmetTipi?.GerekenUzmanlik) continue;

                        bool ustaninCakisanIsiVarMi = tumRandevular.Any(r =>
                            r.Id != arkaRandevu.Id &&
                            r.UstaId == usta.Id &&
                            (r.Durum == RandevuDurumu.Bekliyor || r.Durum == RandevuDurumu.Islemde) &&
                            (r.Tarih < denenenSaat.AddMinutes(islemSuresi) && r.Tarih.AddMinutes(islemSuresi) > denenenSaat)
                        );

                        if (!ustaninCakisanIsiVarMi)
                        {
                            arkaRandevu.UstaId = usta.Id;
                            arkaRandevu.Tarih = denenenSaat;

                            await _randevuRepository.UpdateAsync(arkaRandevu);
                            alternatifBulundu = true;

                            if (denenenSaat > orijinalSaat)
                            {
                                await _publishEndpoint.Publish(new RandevuKaydirildiEvent
                                {
                                    RandevuId = arkaRandevu.Id,
                                    MusteriEmail = arkaRandevu.Kullanici?.Email ?? "musteri@ornek.com", // Null koruması eklendi
                                    MusteriAdSoyad = arkaRandevu.Kullanici?.AdSoyad ?? "Sayın Müşterimiz",
                                    AracPlaka = arkaRandevu.Arac?.Plaka ?? string.Empty, // Null koruması eklendi
                                    EskiTarih = orijinalSaat,
                                    YeniTarih = denenenSaat,
                                    Tip = KaydirmaTipi.HafifGuncelleme,
                                    Aciklama = "Önceki işlemlerin uzamasından dolayı randevunuz güncellenmiştir."
                                });
                            }
                            break;
                        }
                    }

                    if (alternatifBulundu) break;
                    denenenSaat = denenenSaat.AddMinutes(5);
                }

                if (mesaiAsimiOldu)
                {
                    var ertesiGun = arkaRandevu.Tarih.Date.AddDays(1).Add(mesaiBaslangic);
                    if (ertesiGun.DayOfWeek == DayOfWeek.Sunday) ertesiGun = ertesiGun.AddDays(1);

                    arkaRandevu.Tarih = ertesiGun;
                    arkaRandevu.UstaId = randevu.UstaId;

                    await _randevuRepository.UpdateAsync(arkaRandevu);

                    await _publishEndpoint.Publish(new RandevuKaydirildiEvent
                    {
                        RandevuId = arkaRandevu.Id,
                        MusteriEmail = arkaRandevu.Kullanici?.Email ?? "musteri@ornek.com",
                        MusteriAdSoyad = arkaRandevu.Kullanici?.AdSoyad ?? "Sayın Müşterimiz",
                        AracPlaka = arkaRandevu.Arac?.Plaka ?? string.Empty,
                        EskiTarih = orijinalSaat,
                        YeniTarih = ertesiGun,
                        Tip = KaydirmaTipi.VipErtesiGun,
                        Aciklama = "İşlem atölye mesaimizi aştığı için randevunuz ertesi gün sabah 'VIP İlk Araç' statüsüne taşınmıştır."
                    });
                }
                else if (!alternatifBulundu)
                {
                    arkaRandevu.Tarih = orijinalSaat.AddMinutes(dto.EkSureDakika);

                    if (arkaRandevu.Tarih.TimeOfDay < molaBitis && arkaRandevu.Tarih.AddMinutes(islemSuresi).TimeOfDay > molaBaslangic)
                    {
                        arkaRandevu.Tarih = arkaRandevu.Tarih.Date.Add(molaBitis);
                    }

                    await _randevuRepository.UpdateAsync(arkaRandevu);

                    await _publishEndpoint.Publish(new RandevuKaydirildiEvent
                    {
                        RandevuId = arkaRandevu.Id,
                        MusteriEmail = arkaRandevu.Kullanici?.Email ?? "musteri@ornek.com",
                        MusteriAdSoyad = arkaRandevu.Kullanici?.AdSoyad ?? "Sayın Müşterimiz",
                        AracPlaka = arkaRandevu.Arac?.Plaka ?? string.Empty,
                        EskiTarih = orijinalSaat,
                        YeniTarih = arkaRandevu.Tarih,
                        Tip = KaydirmaTipi.KrizErtelemesi,
                        Aciklama = "Tüm birimlerimizdeki acil yoğunluk sebebiyle randevu saatiniz zorunlu olarak kaydırılmıştır."
                    });
                }
            }

            // Performans iyileştirmesi: Tüm güncellemeler bittikten sonra tek seferde veritabanına yazılır 
            // FİNAL: Hangi repodan çağırdığının önemi yok, o anki DbContext'i tetikler! ve Ramde tutulan tüm işlemler(update,delete,vs) tek bir transaction(pakette) artık dbde kaydedilir
            await _randevuRepository.SaveChangesAsync();

            return true;
        }

        public async Task<bool> RandevuBaslatAsync(int randevuId)
        {
            // 1. Randevuyu buluyoruz
            var randevu = await _randevuRepository.GetByIdAsync(randevuId);
            if (randevu == null || randevu.Durum != RandevuDurumu.Bekliyor)
                return false;

            // Randevuyu işlemde olarak işaretle ve gerçek başlama saatini damgala
            randevu.Durum = RandevuDurumu.Islemde;
            randevu.GerceklesenBaslangic = DateTime.Now;

            await _randevuRepository.UpdateAsync(randevu);

            // 2. Eğer randevuya atanmış bir istasyon varsa, veritabanından çekip meşgule alıyoruz
            if (randevu.IstasyonId.HasValue)
            {
                var istasyon = await _istasyonRepository.GetAsync(i => i.Id == randevu.IstasyonId.Value);
                if (istasyon != null)
                {
                    istasyon.MusaitMi = false; // Lifti meşgule (dolu) çekiyoruz
                    await _istasyonRepository.UpdateAsync(istasyon);
                }
            }

            // --- EN KRİTİK NOKTA: DEĞİŞİKLİKLERİ TEK SEFERDE SQL'E KOMPLE YAZIYORUZ ---
            await _randevuRepository.SaveChangesAsync();
            return true;
        }
        public async Task<Fatura> IslemiTamamlaVeFaturaKesAsync(RandevuTamamlamaDto dto)
        {
            var randevu = await _randevuRepository.GetAsync(r => r.Id == dto.RandevuId);
            if (randevu == null) throw new Exception("Randevu bulunamadı.");
            if (randevu.Durum == RandevuDurumu.Tamamlandi) throw new Exception("Bu randevu zaten tamamlanmış.");

            var hizmet = await _hizmetTipiRepository.GetAsync(h => h.Id == randevu.HizmetTipiId);
            decimal iscilikUcreti = hizmet.Fiyat;
            decimal toplamParcaMaliyeti = 0;

            var faturayaGidecekParcalar = new List<YedekParca>();

            foreach (var parcaDto in dto.KullanilanParcalar)
            {
                var stoktakiParca = await _yedekParcaRepository.GetAsync(y => y.Id == parcaDto.YedekParcaId);

                if (stoktakiParca == null) throw new Exception("Yedek parça bulunamadı.");
                if (stoktakiParca.StokMiktari < parcaDto.KullanilanMiktar) throw new Exception($"{stoktakiParca.ParcaAdi} için stok yetersiz.");

                stoktakiParca.StokMiktari -= parcaDto.KullanilanMiktar;
                await _yedekParcaRepository.UpdateAsync(stoktakiParca);

                toplamParcaMaliyeti += (stoktakiParca.BirimFiyat * parcaDto.KullanilanMiktar);

                var randevuParca = new RandevuParca
                {
                    RandevuId = randevu.Id,
                    YedekParcaId = stoktakiParca.Id,
                    KullanilanMiktar = parcaDto.KullanilanMiktar
                };
                await _randevuParcaRepository.AddAsync(randevuParca);
                await _randevuParcaRepository.SaveChangesAsync();

                faturayaGidecekParcalar.Add(stoktakiParca);
            }

            // --- 1. İŞLEM BİTTİĞİ İÇİN RANDEVUYU GÜNCELLİYORUZ ---
            randevu.Durum = RandevuDurumu.Tamamlandi;
            randevu.GerceklesenBitis = DateTime.Now;
            await _randevuRepository.UpdateAsync(randevu);

            // --- 2. KRİTİK DOKUNUŞ: LİFTİ (İSTASYONU) TEKRAR BOŞA (MÜSAİTE) ÇIKARIYORUZ! ---
            if (randevu.IstasyonId.HasValue)
            {
                var istasyon = await _istasyonRepository.GetByIdAsync(randevu.IstasyonId.Value);
                if (istasyon != null)
                {
                    istasyon.MusaitMi = true; // Lifti tekrar MÜSAİT (1 / true) yaptık!
                    await _istasyonRepository.UpdateAsync(istasyon);
                }
            }

            await _randevuRepository.SaveChangesAsync();

            // --- 3. FATURA ÜRETİMİ VE MAİL SÜREÇLERİ ---
            var fatura = new Fatura
            {
                RandevuId = randevu.Id,
                IscilikUcreti = iscilikUcreti,
                ToplamParcaMaliyeti = toplamParcaMaliyeti,
                KesimTarihi = DateTime.Now
            };

            await _faturaRepository.AddAsync(fatura);
            await _faturaRepository.SaveChangesAsync();

            randevu.Kullanici = await _kullaniciRepository.GetAsync(m => m.Id == randevu.KullaniciId);
            randevu.Arac = await _aracRepository.GetAsync(a => a.Id == randevu.AracId);
            randevu.HizmetTipi = hizmet;
            randevu.RandevuParcalari = (await _randevuParcaRepository.GetAllAsync(rp => rp.RandevuId == randevu.Id)).ToList();

            string pdfYolu = await _pdfFaturaService.FaturaUretVeKaydetAsync(fatura, randevu, faturayaGidecekParcalar);

            fatura.PdfDosyaYolu = pdfYolu;
            await _faturaRepository.UpdateAsync(fatura);
            await _faturaRepository.SaveChangesAsync();

            await _publishEndpoint.Publish(new FaturaKesildiEvent
            {
                FaturaId = fatura.Id,
                RandevuId = randevu.Id,
                MusteriEmail = randevu.Kullanici?.Email ?? string.Empty,
                MusteriAdSoyad = randevu.Kullanici?.AdSoyad ?? "Sayın Müşterimiz",
                AracPlaka = randevu.Arac?.Plaka ?? string.Empty,
                ToplamTutar = fatura.GenelToplam,
                KesimTarihi = fatura.KesimTarihi,
                IscilikUcreti = fatura.IscilikUcreti,
                Parca = fatura.ToplamParcaMaliyeti,
                PdfDosyaYolu = pdfYolu
            });

            return fatura;
        }

        public async Task<IEnumerable<Randevu>> GetUstaninGunlukRandevulariAsync(int ustaId)
        {
            var tumRandevular = await _randevuRepository.TumRandevulariIliskileriyleGetirAsync();

            // Sadece BUGÜNE ait ve o USTAYA atanmış randevuları saatine göre sıralayarak getir
            return tumRandevular
                .Where(r => r.UstaId == ustaId && r.Tarih.Date == DateTime.Today)
                .OrderBy(r => r.Tarih)
                .ToList();
        }

        public async Task<IEnumerable<Randevu>> GetGunlukTumRandevularAsync()
        {
            var tumRandevular = await _randevuRepository.TumRandevulariIliskileriyleGetirAsync();

            // Sadece BUGÜNE ait TÜM randevuları saatine göre sıralayarak getir
            return tumRandevular
             .Where(r => r.Tarih.Date == DateTime.Today.Date)
             .OrderBy(r => r.Tarih)
             .ToList();
        }
