                ValidateAudience = true,
                ValidateLifetime = true,
                ValidateIssuerSigningKey = true,
                ValidIssuer = builder.Configuration["JwtSettings:Issuer"],
                ValidAudience = builder.Configuration["JwtSettings:Audience"],
                IssuerSigningKey = new SymmetricSecurityKey(
                    Encoding.UTF8.GetBytes(builder.Configuration["JwtSettings:Key"]!))

            options.Events = new JwtBearerEvents
            {
                OnTokenValidated = async context =>

                    var cache = context.HttpContext.RequestServices.GetRequiredService<IDistributedCache>();
                    var rawToken = context.Request.Headers["Authorization"].ToString().Replace("Bearer ", "").Trim();

                    if (!string.IsNullOrEmpty(rawToken))
                    {
                        var isBlacklisted = await cache.GetStringAsync($"blacklist:{rawToken}");
                        if (!string.IsNullOrEmpty(isBlacklisted))
                        {
                            // Token Redis'te kara listedeyse isteği hemen kapıda reddet
                            context.Fail("Bu oturum sonlandırılmıştır. Lütfen tekrar giriş yapın.");

                    }

            In = ParameterLocation.Header,
            Type = SecuritySchemeType.ApiKey,
            Scheme = "Bearer"
        });
        c.AddSecurityRequirement(new OpenApiSecurityRequirement
        {
        {
            new OpenApiSecurityScheme
            {
                Reference = new OpenApiReference
                {
                    Type = ReferenceType.SecurityScheme,
                    Id = "Bearer"
                }
            },
            Array.Empty<string>()
        }
        });
    });

    builder.Services.AddCors(options =>

        {
            // AllowAnyOrigin YERİNE bunu kullanıyoruz. Hem her şeye izin verir, hem çökmez.
            policy.SetIsOriginAllowed(origin => true)
                  .AllowAnyHeader()
                  .AllowAnyMethod()
                  .AllowCredentials(); // SignalR için mecburi
        });
    });

    //builder.Services.AddCors(options =>

    //        policy.WithOrigins("http://localhost:5173") // Senin frontend portun
    //              .AllowAnyHeader()
    //              .AllowAnyMethod()
    //              .AllowCredentials(); // SignalR bildirimleri için bu satır şarttır!
    //    });
    //});

    builder.Services.AddScoped(typeof(IRepository<>), typeof(Repository<>));
    builder.Services.AddScoped<IRandevuRepository, RandevuRepository>();

using Microsoft.AspNetCore.Authorization;
using Microsoft.AspNetCore.Mvc;
using OtoServisApi.DTOs;
using OtoServisApi.Services;

namespace OtoServisApi.Controllers
{
    [ApiController]
    [Route("api/[controller]")]
    public class AdminController : ControllerBase
    {
        private readonly IAuthService _authService;

        public AdminController(IAuthService authService)
        {
            _authService = authService;
        }

        // --- SADECE SUPERADMIN KULLANABİLİR ---
        [HttpPost("admin-ekle")]
        [Authorize(Roles = "SuperAdmin")]
        public async Task<IActionResult> AdminEkle([FromBody] PersonelEkleDto dto)
        {
            if (!ModelState.IsValid) return BadRequest(ModelState);

            // Güvenlik: DTO içinden gelen Rol verisini görmezden geliyoruz. 
            // SuperAdmin bu ucu kullandığı an eklenecek kişi zorunlu "Admin" olur.
            var sonuc = await _authService.PersonelEkleAsync(dto, "Admin");

            if (!sonuc.BasariliMi) return BadRequest(new { Mesaj = sonuc.Mesaj });

            return Ok(new { Mesaj = "Yeni Admin başarıyla eklendi." });
        }

        // --- ADMIN VE SUPERADMIN KULLANABİLİR ---
        [HttpPost("personel-ekle")]
        [Authorize(Roles = "SuperAdmin,Admin")]
        public async Task<IActionResult> PersonelEkle([FromBody] PersonelEkleDto dto)
        {
            if (!ModelState.IsValid) return BadRequest(ModelState);

            // Güvenlik Kalkanı: Admin, kendi kendine SuperAdmin veya başka bir Admin ekleyemez!
            var yasakliRoller = new List<string> { "SuperAdmin", "Admin" };

            if (yasakliRoller.Any(r => r.Equals(dto.Rol, StringComparison.OrdinalIgnoreCase))) // büyük küçük harf duyarlılığını ortadan kaldırıyoruzz
            if (yasakliRoller.Contains(dto.Rol))
            {
                return BadRequest(new { Mesaj = "Bu yetki seviyesinde bir personel ekleme izniniz yok." });
            }

            // Güvenlikten geçtiyse, kullanıcının belirlediği rolü ("Usta", "Vezne" vb.) servise yolla
            var sonuc = await _authService.PersonelEkleAsync(dto, dto.Rol);
            

            if (!sonuc.BasariliMi) return BadRequest(new { Mesaj = sonuc.Mesaj });

            return Ok(new { Mesaj = $"{dto.Rol} rolündeki personel başarıyla eklendi." });
        }
    }
}

using Microsoft.AspNetCore.Authorization;
using Microsoft.AspNetCore.Mvc;
using Microsoft.Extensions.Caching.Distributed;
using OtoServisApi.DTOs;
using OtoServisApi.Services;
using System.IdentityModel.Tokens.Jwt;

namespace OtoServisApi.Controllers
{
    [Route("api/[controller]")]
    [ApiController]
    public class AuthController : ControllerBase
    {
        private readonly IAuthService _authService;

        public AuthController(IAuthService authService)
        {
            _authService = authService;
        }

        [HttpPost("register")]
        public async Task<IActionResult> Register([FromBody] RegisterDto dto)
        {
            try
            {
                var sonuc = await _authService.RegisterAsync(dto);
                return Ok(new { Mesaj = sonuc });
            }
            catch (Exception ex)
            {
                return BadRequest(new { Hata = ex.Message });
            }
        }

        [HttpPost("login")]
        public async Task<IActionResult> Login([FromBody] LoginDto dto)
        {
            try
            {
                var token = await _authService.LoginAsync(dto);
                return Ok(new { Token = token });
            }
            catch (Exception ex)
            {
                return BadRequest(new { Hata = ex.Message });
            }
        }

        [Authorize]
        [HttpPost("logout")]
        public async Task<IActionResult> Logout([FromServices] IDistributedCache cache)
        {
            // 1. İstekten gelen ham Bearer token'ı al
            var rawToken = Request.Headers["Authorization"].ToString().Replace("Bearer ", "").Trim();
            if (string.IsNullOrEmpty(rawToken))
                return BadRequest("Token bulunamadı.");

            // 2. Token'ın süresinin bitmesine kaç dakika/saat kaldığını hesapla
            var handler = new JwtSecurityTokenHandler();
            var jwtToken = handler.ReadJwtToken(rawToken);
            var kalanSure = jwtToken.ValidTo - DateTime.UtcNow;

            // Eğer süre zaten dolmuşsa işlem yapmaya gerek yok
            if (kalanSure <= TimeSpan.Zero)
                return Ok(new { Mesaj = "Oturum zaten sonlanmış." });

            // 3. Token'ı kalan ömrü kadar Redis'e 'kara liste' olarak kaydet
            var cacheKey = $"blacklist:{rawToken}";
            var cacheOptions = new DistributedCacheEntryOptions
            {
                AbsoluteExpirationRelativeToNow = kalanSure
            };

            await cache.SetStringAsync(cacheKey, "revoked", cacheOptions);

            return Ok(new { Mesaj = "Başarıyla çıkış yapıldı. Token geçersiz kılındı." });
        }

        [HttpPost("sifremi-unuttum")]
        public async Task<IActionResult> SifremiUnuttum([FromBody] SifremiUnuttumDto dto)
        {
            if (!ModelState.IsValid) return BadRequest(ModelState);

            var sonuc = await _authService.SifreSifirlamaKoduGonderAsync(dto);
            return Ok(new { Mesaj = sonuc.Mesaj });
        }


        [HttpPost("sifre-sifirla")]
        public async Task<IActionResult> SifreSifirla([FromBody] SifreSifirlaDto dto)
        {
            if (!ModelState.IsValid) return BadRequest(ModelState);

            var sonuc = await _authService.YeniSifreyiKaydetAsync(dto);
            if (!sonuc.BasariliMi)
            {
                return BadRequest(new { Mesaj = sonuc.Mesaj });
            }

            return Ok(new { Mesaj = sonuc.Mesaj });
        }
    }
}

using Microsoft.AspNetCore.Authorization;
using Microsoft.AspNetCore.Mvc;
using Microsoft.Extensions.Caching.Distributed;
using OtoServis.Data.Models;
using OtoServis.Data.Repositories;
using System.Text.Json;

namespace OtoServisApi.Controllers
{
    [Route("api/[controller]")]
    [ApiController]
    public class HizmetTipleriController : ControllerBase
    {
        private readonly IRepository<HizmetTipi> _hizmetTipiRepository;
        private readonly IDistributedCache cache;

        public HizmetTipleriController(IRepository<HizmetTipi> hizmetTipiRepository, IDistributedCache cache)
         
        {
            _hizmetTipiRepository = hizmetTipiRepository;
            cache = cache;
        }

        [HttpGet]
        //[Authorize(Roles = "SuperAdmin,Admin")]
        [AllowAnonymous]
        public async Task<ActionResult<IEnumerable<HizmetTipi>>> GetHizmetTipleri()
        {
            string cacheKey = "tum_hizmetler";

            // 1. Önce Redis'e bak
            var cachedData = await cache.GetStringAsync(cacheKey);
            if (!string.IsNullOrEmpty(cachedData))
            {
                var jsonHizmetler = JsonSerializer.Deserialize<List<HizmetTipi>>(cachedData);
                return Ok(jsonHizmetler);
            }

            // 2. Redis'te yoksa SQL Server'dan getir
            var hizmetler = (await _hizmetTipiRepository.GetAllAsync()).ToList();

            // 3. Redis'e 30 dakikalığına kaydet
            var cacheOptions = new DistributedCacheEntryOptions
            {
                AbsoluteExpirationRelativeToNow = TimeSpan.FromMinutes(30)
            };
            await cache.SetStringAsync(cacheKey, JsonSerializer.Serialize(hizmetler), cacheOptions);

            return Ok(hizmetler);
        }

        [HttpPost]
        [Authorize(Roles = "SuperAdmin,Admin")]
        public async Task<ActionResult<HizmetTipi>> PostHizmetTipi(HizmetTipi hizmetTipi, [FromServices] IDistributedCache cache)
        {
            await _hizmetTipiRepository.AddAsync(hizmetTipi);
            await _hizmetTipiRepository.SaveChangesAsync();

            // ÖNBELLEĞİ BOŞALT: Yeni hizmet eklendiği için eski listeyi çöpe atıyoruz
            await cache.RemoveAsync("tum_hizmetler");

            return CreatedAtAction(nameof(GetHizmetTipleri), new { id = hizmetTipi.Id }, hizmetTipi);
        }
    }
}

using Microsoft.AspNetCore.Authorization;
using Microsoft.AspNetCore.Mvc;
using OtoServisApi.DTOs;
using OtoServis.Data.Models;
using OtoServis.Data.Repositories;
using System.Security.Claims;

namespace OtoServisApi.Controllers
{
    [Route("api/[controller]")]
    [ApiController]
    [Authorize] // Token'ı olan (müşteri) ödeme yapabilir
    public class OdemeController : ControllerBase
    {
        private readonly IRepository<Fatura> _faturaRepository;
        private readonly IRepository<Randevu> _randevuRepository;


        public OdemeController(IRepository<Fatura> faturaRepository, IRepository<Randevu> randevuRepository)
        {
            _faturaRepository = faturaRepository;
            _randevuRepository = randevuRepository;
        }
               

        [Authorize(Roles = "Veznedar,SuperAdmin")]
        [HttpPost("{faturaId}/fiziksel-ode")]
        public async Task<IActionResult> FizikselOdemeAl(int faturaId, [FromQuery] string odemeTuru) // "Nakit", "POS", "Havale"
        {
            var fatura = await _faturaRepository.GetAsync(f => f.Id == faturaId);

            if (fatura == null) return NotFound("Fatura bulunamadı.");
            if (fatura.OdendiMi) return BadRequest("Bu fatura zaten ödenmiş.");
                        
            fatura.OdendiMi = true;
            fatura.OdemeTarihi = DateTime.Now;
            fatura.OdemeYontemi = odemeTuru; // "Nakit" veya "POS"
            fatura.TahsilEdenKullaniciId = int.Parse(User.FindFirst(ClaimTypes.NameIdentifier)!.Value);

            await _faturaRepository.UpdateAsync(fatura);
            await _faturaRepository.SaveChangesAsync();

            return Ok(new
            {
                Mesaj = $"Ödeme veznede ({odemeTuru}) olarak tahsil edildi. Fatura kapatıldı.",
                FaturaId = fatura.Id,
                Tahsilatci = User.Identity?.Name ?? "Veznedar"
            });
        }


        [Authorize(Roles = "Veznedar,SuperAdmin")]
        [HttpPut("{id}/teslim-et")]
        public async Task<IActionResult> AraciTeslimEt(int id)
        {
            var randevu = await _randevuRepository.GetAsync(r => r.Id == id);
            if (randevu == null) return NotFound("Randevu bulunamadı.");

            if (randevu.Durum != RandevuDurumu.Tamamlandi)
            {
                return BadRequest("Usta henüz bakım/onarım işlemlerini tamamlamadı.");
            }

            var fatura = await _faturaRepository.GetAsync(f => f.RandevuId == id);
            if (fatura == null || !fatura.OdendiMi)
            {
                return BadRequest("Ödeme onayı bulunamadı! Fatura tahsil edilmeden araç teslim edilemez.");
            }

            randevu.TeslimTarihi = DateTime.Now;
            randevu.TeslimEdenKullaniciId = int.Parse(User.FindFirst(ClaimTypes.NameIdentifier)!.Value);
            randevu.Durum = RandevuDurumu.Tamamlandi; // veya RandevuDurumu.TeslimEdildi
            await _randevuRepository.SaveChangesAsync();
            return Ok(new
            {

                Mesaj = "Ödeme kontrolü tamamlandı. Araç ve anahtar müşteriye teslim edilebilir.",
                RandevuId = id
            });
        }
    }
}

            _randevuService = randevuService;
   
            _kullaniciService = kullaniciService;
            _aracRepository = aracRepository;
            _authService = authService;
            _publishEndpoint = publishEndpoint;

        }

        //// GET: api/Randevular
        //[HttpGet]
        //public async Task<ActionResult<IEnumerable<RandevuDetayDto>>> GetRandevular()
        //{

        [Authorize(Roles = "Musteri")]
        [HttpPost("Kayıtlı Müşteri Randevu")]
        public async Task<IActionResult> KayitliYeniRandevu([FromBody] KayitliRandevuTalebiDto kisaDto)
        {
            var musteriId = int.Parse(User.FindFirst(System.Security.Claims.ClaimTypes.NameIdentifier)!.Value);
            var guncelKullanici = await _kullaniciService.GetKullaniciBilgileriAsync(musteriId);


            string sonMarkaModel = kisaDto.AracMarkaModel;
            int sonUretimYili = kisaDto.AracUretimYili ?? 0;

            if (string.IsNullOrEmpty(kisaDto.AracMarkaModel))
            {
                var kayitliAracList = await _aracRepository.GetAllAsync(a => a.Plaka == kisaDto.AracPlaka);
                var KayitliArac = kayitliAracList.FirstOrDefault();
                if (KayitliArac != null)
                {
                    sonMarkaModel = KayitliArac.MarkaModel;
                    sonUretimYili = KayitliArac.UretimYili;
                }
            }

            
            var anaDto = new RandevuOlusturDto
            {
                MusteriAdSoyad = guncelKullanici.AdSoyad,
                MusteriTelefon = guncelKullanici.Telefon ?? "Belirtilmemiş",
                AracPlaka = kisaDto.AracPlaka,
                AracMarkaModel = sonMarkaModel,
                AracUretimYili = sonUretimYili,
                Tarih = kisaDto.Tarih,
                MusteriNotu = kisaDto.MusteriNotu,
                HizmetTipiId = 1 
            };

            int randevuID = await _randevuService.RandevuOlusturAsync(anaDto);

            if (randevuID < 0) return BadRequest("Randevu oluşturulamadı.");

            // 3. İŞTE RABBITMQ PAKETİMİZİ (EVENT) HAZIRLIYORUZ
            var aiEvent = new AiAnalizIstendiEvent
            {
                RandevuId = randevuID, // BÖYLECE WORKER HANGİ RANDEVUYU GÜNCELLEYECEĞİNİ BİLECEK!
                AracPlaka = kisaDto.AracPlaka,
                MusteriNotu = kisaDto.MusteriNotu,
                SignalRConnectionId = kisaDto.SignalRConnectionId // Frontend'den gelen bağlantı ID'si
            };

            // 4. EVENTİ KUYRUĞA FIRLATIYORUZ (MassTransit kullanıyorsan)
            await _publishEndpoint.Publish(aiEvent);

            // 5. VE MÜŞTERİYE ANINDA CEVAP DÖNÜYORUZ! (202 Accepted)
            return Accepted(new
            {
                Mesaj = $"Randevunuz başarıyla oluşturuldu (No: {randevuID}). Yapay zeka arıza notunuzu inceliyor, lütfen ekranda kalın...",
                RandevuId = randevuID
            });
        }

        [AllowAnonymous]
        [HttpPost("Misafir Randevu")]
        public async Task<IActionResult> MisafirYeniRandevu(RandevuOlusturDto dto)
        {

            // 1. AuthService'i çağırıyoruz. Sistem adamı ya bulacak ya da yeni ID üretecek.
            int musteriId = await _authService.MisafirKaydetVeyaBulAsync(dto.MusteriAdSoyad, dto.Email, dto.MusteriTelefon);

            // 2. Dönen ID'yi, asıl DTO'muzun içine yerleştiriyoruz (Foreign Key hatası almamak için).
            dto.MusteriId = musteriId;

            // 3. Randevuyu veritabanına kaydediyoruz.
            int randevuID = await _randevuService.RandevuOlusturAsync(dto);


            // 4. Sonuçları kontrol edip müşteriye cevap dönüyoruz.
            if (randevuID < 0)
            {
                return BadRequest("Randevu oluşturulamadı. Lütfen geçerli bir tarih seçiniz.");
            }

            var aiEvent = new AiAnalizIstendiEvent
            {
                RandevuId = randevuID, // BÖYLECE WORKER HANGİ RANDEVUYU GÜNCELLEYECEĞİNİ BİLECEK!
                AracPlaka = dto.AracPlaka,
                MusteriNotu = dto.MusteriNotu,
                SignalRConnectionId = dto.SignalRConnectionId // Frontend'den gelen bağlantı ID'si
            };

            // 4. EVENTİ KUYRUĞA FIRLATIYORUZ (MassTransit kullanıyorsan)
            await _publishEndpoint.Publish(aiEvent);

            // 5. VE MÜŞTERİYE ANINDA CEVAP DÖNÜYORUZ! (202 Accepted)
            return Accepted(new
            {
                Mesaj = $"Randevunuz başarıyla oluşturuldu (No: {randevuID}). Yapay zeka arıza notunuzu inceliyor, lütfen ekranda kalın...",
                RandevuId = randevuID
            });


            return Ok($"Randevu talebiniz başarıyla alındı. Randevu Numaranız: {randevuID}");
        }


        [Authorize(Roles = "Musteri")]
        [HttpGet("benim-randevularim")]
        public async Task<IActionResult> BenimRandevularimiGetir()
        {
            // 1. Kimliği Token'dan Çek
            var kullaniciIdClaim = User.FindFirst(ClaimTypes.NameIdentifier);
            if (kullaniciIdClaim == null) return Unauthorized();

            int kullaniciId = int.Parse(kullaniciIdClaim.Value);

            // 2. İşi Service'e Devret
            var randevular = await _randevuService.GetMusteriRandevulariAsync(kullaniciId);

            // 3. Sonucu Dön
            if (randevular == null || !randevular.Any())
            {
                return Ok(new { Mesaj = "Henüz alınmış bir randevunuz bulunmamaktadır." });
            }

            return Ok(randevular);
        }

using Microsoft.AspNetCore.Authorization;
using Microsoft.AspNetCore.Mvc;
using Microsoft.Extensions.Caching.Distributed;
using OtoServis.Data.Models;
using OtoServis.Data.Repositories;
using System.Text.Json;
using System.Text.Json.Serialization;

namespace OtoServisApi.Controllers
{
    [Route("api/[controller]")]
    [ApiController]
    // Sadece SuperAdmin, Admin ve yeni ekleyeceğimiz DepoSorumlusu girebilir!
    [Authorize(Roles = "SuperAdmin,Admin,DepoSorumlusu")]
    public class YedekParcalarController : ControllerBase
    {
        private readonly IRepository<YedekParca> _yedekParcaRepository;
        private readonly IDistributedCache _cache;

        public YedekParcalarController(IRepository<YedekParca> yedekParcaRepository, IDistributedCache cache)
        {
            _yedekParcaRepository = yedekParcaRepository;
            _cache = cache;
        }

        [HttpGet("tum-stoklar")]
        public async Task<IActionResult> GetYedekParcalar()
        {
            string cacheKey = "tum_yedek_parcalar";

            // 1. Önce Redis'e bak (Cache Hit)
            var cachedData = await _cache.GetStringAsync(cacheKey);
            if (!string.IsNullOrEmpty(cachedData))
            {
                var jsonParcalar = JsonSerializer.Deserialize<List<YedekParca>>(cachedData);
                return Ok(jsonParcalar);
            }

            // 2. Redis'te yoksa veritabanından çek (Cache Miss)
            var parcalar = (await _yedekParcaRepository.GetAllAsync()).ToList();

            // 3. Veriyi 60 dakika geçerli olacak şekilde Redis'e yaz
            var cacheOptions = new DistributedCacheEntryOptions
            {
                AbsoluteExpirationRelativeToNow = TimeSpan.FromMinutes(60)
            };
            await _cache.SetStringAsync(cacheKey, JsonSerializer.Serialize(parcalar), cacheOptions);

            return Ok(parcalar);
        }

        [HttpGet("kritik-stoklar")]
        public async Task<IActionResult> KritikStoklariGetir()
        {
            // Stok miktarı, kritik seviyenin altına düşenleri bul (Ön yüzdeki kırmızı uyarı ekranı için)
            var kritikParcalar = await _yedekParcaRepository.GetAllAsync(p => p.StokMiktari <= p.KritikStokSeviyesi);

            if (!kritikParcalar.Any()) return Ok(new { Mesaj = "Harika! Kritik seviyede hiçbir parça yok." });

            return Ok(kritikParcalar);
        }

        //YENİ PARÇA EKLENDİĞİNDE
        [HttpPost]
        [Authorize(Roles = "SuperAdmin,Admin,DepoSorumlusu")]
        public async Task<ActionResult<YedekParca>> PostYedekParca(YedekParca yedekParca, [FromServices] IDistributedCache cache)
        {
            await _yedekParcaRepository.AddAsync(yedekParca);
            await _yedekParcaRepository.SaveChangesAsync();

            // Önbelleği temizle
            await cache.RemoveAsync("tum_yedek_parcalar");

            return CreatedAtAction(nameof(GetYedekParcalar), new { id = yedekParca.Id }, yedekParca);
        }

        //PARÇA/ STOK GÜNCELLENDİĞİNDE
        [HttpPut("{id}")]
        [Authorize(Roles = "SuperAdmin,Admin,DepoSorumlusu")]
        public async Task<IActionResult> PutYedekParca(int id, YedekParca yedekParca, [FromServices] IDistributedCache cache)
        {
            if (id != yedekParca.Id) return BadRequest("ID uyuşmazlığı.");

            await _yedekParcaRepository.UpdateAsync(yedekParca);
            await _yedekParcaRepository.SaveChangesAsync();

            // Güncel stok/fiyat bilgisinin yansıması için önbelleği temizle
            await cache.RemoveAsync("tum_yedek_parcalar");

            return NoContent();
        }

        //PARÇA SİLİNDİĞİNDE
        [HttpDelete("{id}")]
        [Authorize(Roles = "SuperAdmin,Admin")]
        public async Task<IActionResult> DeleteYedekParca(int id, [FromServices] IDistributedCache cache)
        {
            var parca = await _yedekParcaRepository.GetByIdAsync(id);
            if (parca == null) return NotFound();

            await _yedekParcaRepository.DeleteAsync(parca);
            await _yedekParcaRepository.SaveChangesAsync();

            // Silinen parça listeden düşsün diye önbelleği temizle
            await cache.RemoveAsync("tum_yedek_parcalar");

            return NoContent();
        }
    }
}
