import { BrowserRouter, Navigate, Route, Routes } from 'react-router-dom';
import { AuthProvider } from '@/contexts/AuthProvider';
import { AppProviders } from '@/contexts/AppProviders';
import { Layout } from '@/components/Layout';
import { StaffLayout } from '@/components/StaffLayout';
import { ProtectedRoute } from '@/components/ProtectedRoute';
import { LandingPage } from '@/pages/LandingPage';
import { ServicesPage } from '@/pages/ServicesPage';
import { GuestAppointmentPage } from '@/pages/GuestAppointmentPage';
import { CustomerAppointmentPage } from '@/pages/CustomerAppointmentPage';
import { LoginPage } from '@/pages/LoginPage';
import { RegisterPage } from '@/pages/RegisterPage';
import { ForgotPasswordPage } from '@/pages/ForgotPasswordPage';
import { ResetPasswordPage } from '@/pages/ResetPasswordPage';
import { MyAppointmentsPage } from '@/pages/MyAppointmentsPage';
import { AdminPanelPage } from '@/pages/AdminPanelPage';
import { UstaPanelPage } from '@/pages/UstaPanelPage';
import { VeznedarPanelPage } from '@/pages/VeznedarPanelPage';
import { DepoPanelPage } from '@/pages/DepoPanelPage';
import { ProfilePage } from '@/pages/ProfilePage'; // (Dosya yolunun projene uygun olduğundan emin ol)

export default function App() {
  return (
    <BrowserRouter>
      <AuthProvider>
        <AppProviders>
          <Routes>
            <Route element={<Layout />}>
              <Route path="/" element={<LandingPage />} />
              <Route path="/hizmetler" element={<ServicesPage />} />
              <Route path="/randevu" element={<GuestAppointmentPage />} />
              <Route path="/giris" element={<LoginPage />} />
              <Route path="/kayit" element={<RegisterPage />} />
              <Route path="/sifremi-unuttum" element={<ForgotPasswordPage />} />
              <Route path="/sifre-sifirla" element={<ResetPasswordPage />} />
              <Route
                path="/randevu-olustur"
                element={
                  <ProtectedRoute roles={['Musteri']}>
                    <CustomerAppointmentPage />
                  </ProtectedRoute>
                }
              />
              <Route
                path="/randevularim"
                element={
                  <ProtectedRoute roles={['Musteri']}>
                    <MyAppointmentsPage />
                  </ProtectedRoute>
                }
              />
            </Route>

            <Route
              element={
                <ProtectedRoute roles={['SuperAdmin', 'Admin', 'Usta', 'Veznedar', 'DepoSorumlusu']}>
                  <StaffLayout />
                </ProtectedRoute>
              }
            >
              <Route
                path="/panel"
                element={
                  <ProtectedRoute roles={['SuperAdmin', 'Admin']}>
                    <AdminPanelPage />
                  </ProtectedRoute>
                }
              />
              <Route
                path="/usta"
                element={
                  <ProtectedRoute roles={['SuperAdmin', 'Admin', 'Usta']}>
                    <UstaPanelPage />
                  </ProtectedRoute>
                }
              />
              <Route
                path="/vezne"
                element={
                  <ProtectedRoute roles={['SuperAdmin', 'Veznedar']}>
                    <VeznedarPanelPage />
                  </ProtectedRoute>
                }
              />
              <Route
                path="/depo"
                element={
                  <ProtectedRoute roles={['SuperAdmin', 'Admin', 'DepoSorumlusu', 'Usta']}>
                    <DepoPanelPage />
                  </ProtectedRoute>
                }
              />

                  <Route path="/profil" element={
              <ProtectedRoute roles={['Musteri']}>
                <ProfilePage />
              </ProtectedRoute>
                  } />
            </Route>

            <Route path="/admin" element={<Navigate to="/panel" replace />} />
            <Route path="*" element={<Navigate to="/" replace />} />
          </Routes>
        </AppProviders>
      </AuthProvider>
    </BrowserRouter>
  );
}

import { Wrench, Calendar, LayoutGrid, LogOut, LogIn, User, Menu, X, Shield } from 'lucide-react';
import { Link, useLocation, useNavigate } from 'react-router-dom';
import { useEffect, useState } from 'react';
import { useAuth } from '@/contexts/AuthContext';
import { dashboardPath } from '@/lib/jwt';

export function Navbar() {
  const { user, logout, isAuthenticated } = useAuth();
  const location = useLocation();
  const navigate = useNavigate();
  const [mobileOpen, setMobileOpen] = useState(false);
  const [scrolled, setScrolled] = useState(false);

  useEffect(() => {
    const onScroll = () => setScrolled(window.scrollY > 8);
    window.addEventListener('scroll', onScroll);
    return () => window.removeEventListener('scroll', onScroll);
  }, []);

  useEffect(() => {
    setMobileOpen(false);
  }, [location.pathname]);

  const handleLogout = async () => {
    await logout();
    navigate('/');
  };

  // --- KESİN VE NET ROL KONTROLLERİ ---
  // 1. Ziyaretçi: Giriş yapmamış VEYA arka planda yanlışlıkla Misafir tokeni almış kişiler
  const isGuest = !isAuthenticated || user?.Rol === 'Misafir' || user?.Rol === 'misafir';
  
  // 2. Kayıtlı Müşteri
  const isMusteri = isAuthenticated && user?.Rol === 'Musteri';
  
  // 3. Personel (Admin, Usta, Vezne vb.)
  const isStaff = isAuthenticated && !isGuest && !isMusteri;
  
  const panel = dashboardPath(user?.Rol);

  const isActive = (path: string) =>
    path === '/' ? location.pathname === '/' : location.pathname.startsWith(path);

  // Linkler için ortak stil tanımları (Kod tekrarını önlemek için)
  const linkBaseClass = "flex items-center gap-2 rounded-lg px-3 py-2 text-sm font-medium transition-all";
  const mobileLinkBaseClass = "flex items-center gap-2 rounded-lg px-3 py-2.5 text-sm font-medium transition-all";
  const activeClass = "bg-brand-50 text-brand-700";
  const inactiveClass = "text-ink-600 hover:bg-ink-100 hover:text-ink-900";

  return (
    <header
      className={`fixed top-0 left-0 right-0 z-50 transition-all duration-300 ${
        scrolled ? 'glass shadow-sm border-b border-ink-200/60' : 'bg-transparent'
      }`}
    >
      <nav className="mx-auto max-w-7xl px-4 sm:px-6 lg:px-8">
        <div className="flex h-16 items-center justify-between">
          
          {/* SOL: LOGO */}
          <Link to="/" className="flex items-center gap-2 group">
            <div className="flex items-center justify-center w-10 h-10 rounded-xl bg-gradient-to-br from-brand-600 to-brand-800 shadow-lg shadow-brand-600/20 group-hover:scale-105 transition-transform">
              <Wrench className="h-5 w-5 text-white" />
            </div>
            <div className="hidden sm:block">
              <span className="font-display font-bold text-ink-900 text-lg leading-none">OtoServis</span>
              <span className="block text-xs text-ink-500 leading-none mt-0.5">Akıllı Bakım Platformu</span>
            </div>
          </Link>

          {/* ORTA: DİNAMİK LİNKLER (MASAÜSTÜ) */}
          <div className="hidden md:flex items-center gap-1">
            <Link to="/" className={`${linkBaseClass} ${isActive('/') ? activeClass : inactiveClass}`}>
              <LayoutGrid className="h-4 w-4" /> Ana Sayfa
            </Link>
            
            <Link to="/hizmetler" className={`${linkBaseClass} ${isActive('/hizmetler') ? activeClass : inactiveClass}`}>
              <Wrench className="h-4 w-4" /> Hizmetler
            </Link>

            {/* Sadece Ziyaretçiler Görebilir */}
            {isGuest && (
              <Link to="/randevu" className={`${linkBaseClass} ${isActive('/randevu') ? activeClass : inactiveClass}`}>
                <Calendar className="h-4 w-4" /> Misafir Randevu
              </Link>
            )}

            {/* Sadece Kayıtlı Müşteriler Görebilir */}
            {isMusteri && (
              <>
                <Link to="/randevu-olustur" className={`${linkBaseClass} ${isActive('/randevu-olustur') ? activeClass : inactiveClass}`}>
                  <Calendar className="h-4 w-4" /> Randevu Al
                </Link>
                <Link to="/randevularim" className={`${linkBaseClass} ${isActive('/randevularim') ? activeClass : inactiveClass}`}>
                  <Calendar className="h-4 w-4" /> Randevularım
                </Link>
              </>
            )}

            {/* Sadece Personeller Görebilir */}
            {isStaff && (
              <Link to={panel} className={`${linkBaseClass} ${isActive(panel) ? activeClass : inactiveClass}`}>
                <Shield className="h-4 w-4" /> Panel
              </Link>
            )}
          </div>

          {/* SAĞ: GİRİŞ / PROFİL BÖLÜMÜ (MASAÜSTÜ) */}
          <div className="hidden md:flex items-center gap-2">
            {(isMusteri || isStaff) ? (
              <>
                <Link to="/profil" className="flex items-center gap-2 px-3 py-1.5 rounded-lg bg-ink-100 hover:bg-ink-200 transition-colors cursor-pointer">
                  <div className="flex items-center justify-center w-7 h-7 rounded-full bg-brand-600 text-white text-xs font-semibold shadow-sm">
                    {(user?.AdSoyad || user?.Rol || 'U').charAt(0).toUpperCase()}
                  </div>
                  <span className="text-sm font-medium text-ink-700 max-w-[140px] truncate">
                    {user?.AdSoyad || user?.Rol}
                  </span>
                </Link>
                <button onClick={handleLogout} className="btn-ghost text-error-600 hover:bg-error-50 hover:text-error-700">
                  <LogOut className="h-4 w-4" /> Çıkış
                </button>
              </>
            ) : (
              <>
                <Link to="/giris" className="btn-ghost">
                  <LogIn className="h-4 w-4" /> Giriş
                </Link>
                <Link to="/kayit" className="btn-primary">
                  <User className="h-4 w-4" /> Kayıt Ol
                </Link>
              </>
            )}
          </div>

          {/* MOBİL MENÜ BUTONU */}
          <button
            className="md:hidden p-2 rounded-lg text-ink-700 hover:bg-ink-100"
            onClick={() => setMobileOpen((v) => !v)}
          >
            {mobileOpen ? <X className="h-5 w-5" /> : <Menu className="h-5 w-5" />}
          </button>
        </div>

        {/* MOBİL AÇILIR MENÜ */}
        {mobileOpen && (
          <div className="md:hidden pb-4 animate-fade-in">
            <div className="flex flex-col gap-1">
              <Link to="/" className={`${mobileLinkBaseClass} ${isActive('/') ? activeClass : inactiveClass}`}>
                <LayoutGrid className="h-4 w-4" /> Ana Sayfa
              </Link>
              
              <Link to="/hizmetler" className={`${mobileLinkBaseClass} ${isActive('/hizmetler') ? activeClass : inactiveClass}`}>
                <Wrench className="h-4 w-4" /> Hizmetler
              </Link>

              {isGuest && (
                <Link to="/randevu" className={`${mobileLinkBaseClass} ${isActive('/randevu') ? activeClass : inactiveClass}`}>
                  <Calendar className="h-4 w-4" /> Misafir Randevu
                </Link>
              )}

              {isMusteri && (
                <>
                  <Link to="/randevu-olustur" className={`${mobileLinkBaseClass} ${isActive('/randevu-olustur') ? activeClass : inactiveClass}`}>
                    <Calendar className="h-4 w-4" /> knk ne alaka
                  </Link>
                  <Link to="/randevularim" className={`${mobileLinkBaseClass} ${isActive('/randevularim') ? activeClass : inactiveClass}`}>
                    <Calendar className="h-4 w-4" /> Randevularım
                  </Link>
                </>
              )}

              {isStaff && (
                <Link to={panel} className={`${mobileLinkBaseClass} ${isActive(panel) ? activeClass : inactiveClass}`}>
                  <Shield className="h-4 w-4" /> Panel
                </Link>
              )}

              <div className="mt-2 pt-2 border-t border-ink-200">
                {(isMusteri || isStaff) ? (
                  <>
                    <Link to="/profil" className="w-full btn-ghost justify-start mb-1 text-ink-700">
                      <User className="h-4 w-4" /> Profilim
                    </Link>
                    <button onClick={handleLogout} className="w-full btn-ghost justify-start text-error-600 hover:bg-error-50">
                      <LogOut className="h-4 w-4" /> Çıkış Yap
                    </button>
                  </>
                ) : (
                  <div className="flex gap-2 px-1">
                    <Link to="/giris" className="flex-1 btn-secondary justify-center">Giriş</Link>
                    <Link to="/kayit" className="flex-1 btn-primary justify-center">Kayıt Ol</Link>
                  </div>
                )}
              </div>
            </div>
          </div>
        )}
      </nav>
    </header>
  );
}

import { Navigate, useLocation } from 'react-router-dom';
import { useAuth } from '@/contexts/AuthContext';
import { LoadingSpinner } from './Loading';

export function ProtectedRoute({
  children,
  roles,
}: {
  children: React.ReactNode;
  roles?: string[];
}) {
  const { isAuthenticated, loading, user } = useAuth();
  const location = useLocation();

  if (loading) return <LoadingSpinner fullPage label="Oturum kontrol ediliyor..." />;
  if (!isAuthenticated) {
    return <Navigate to="/giris" replace state={{ from: location.pathname }} />;
  }
  if (roles && user?.Rol && !roles.includes(String(user.Rol))) {
    return <Navigate to="/" replace />;
  }
  return <>{children}</>;
}

import { useCallback, useState } from 'react';
import {

  XCircle,

export function ToastProvider({ children }: { children: React.ReactNode }) {
  const [toasts, setToasts] = useState<Toast[]>([]);

  const dismiss = useCallback((id: string) => {
    setToasts((prev) => prev.filter((t) => t.id !== id));
  }, []);

  const show = useCallback((toast: Omit<Toast, 'id'>): string => {
    const id = `${Date.now()}-${Math.random().toString(36).slice(2)}`;
    const duration = toast.duration ?? (toast.type === 'ai' ? 12000 : 5000);
    const newToast: Toast = { ...toast, id, duration };
    setToasts((prev) => [...prev, newToast]);
    if (duration > 0) {
      setTimeout(() => dismiss(id), duration);
    }
    return id;
  }, [dismiss]);

  const showSuccess = useCallback((title: string, message?: string) =>
    show({ type: 'success', title, message }), [show]);

  const showError = useCallback((title: string, message?: string) =>
    show({ type: 'error', title, message }), [show]);

  const showInfo = useCallback((title: string, message?: string) =>
    show({ type: 'info', title, message }), [show]);

  const showWarning = useCallback((title: string, message?: string) =>
    show({ type: 'warning', title, message }), [show]);

  const showAi = useCallback((data: AiAnalysisMessage) => {
    const title = data.Baslik || 'Yapay Zeka Analiz Sonucu';
    const message = data.Icerik || data.raw || '';
    show({
      type: 'ai',
      title,
      message,
      aiData: data,
      duration: 15000,
    });
  }, [show]);

  return (
    <ToastContext.Provider
      value={{ toasts, show, dismiss, showSuccess, showError, showInfo, showWarning, showAi }}
    >
      {children}
      <ToastViewport toasts={toasts} onDismiss={dismiss} />
    </ToastContext.Provider>
  );
}

function ToastViewport({
  toasts,
  onDismiss,
}: {
  toasts: Toast[];
  onDismiss: (id: string) => void;
}) {
  return (
    <div className="fixed bottom-4 right-4 z-[9999] flex max-w-sm flex-col gap-3 pointer-events-none">
      {toasts.map((toast) => (
        <ToastCard key={toast.id} toast={toast} onDismiss={() => onDismiss(toast.id)} />
      ))}
    </div>
  );
}

function ToastCard({ toast, onDismiss }: { toast: Toast; onDismiss: () => void }) {
  const [exiting, setExiting] = useState(false);

  const handleClose = () => {
    setExiting(true);
    setTimeout(onDismiss, 300);
  };

  if (toast.type === 'ai') {
    return <AiToastCard toast={toast} onDismiss={handleClose} exiting={exiting} />;
  }

  const config: Record<ToastType, { icon: typeof Info; bg: string; border: string; iconColor: string }> = {
    success: { icon: CheckCircle2, bg: 'bg-white', border: 'border-l-4 border-l-success-500', iconColor: 'text-success-500' },
    error: { icon: XCircle, bg: 'bg-white', border: 'border-l-4 border-l-error-500', iconColor: 'text-error-500' },
    info: { icon: Info, bg: 'bg-white', border: 'border-l-4 border-l-brand-500', iconColor: 'text-brand-500' },
    warning: { icon: AlertTriangle, bg: 'bg-white', border: 'border-l-4 border-l-warning-500', iconColor: 'text-warning-500' },
    ai: { icon: Sparkles, bg: 'bg-white', border: 'border-l-4 border-l-accent-500', iconColor: 'text-accent-500' },

import { useCallback, useEffect, useState } from 'react';
import { AuthContext, type AuthContextValue } from './AuthContext';
import { authApi, clearAuth, getStoredUser } from '@/services/api';
import { pickToken } from '@/lib/json';
import { userFromToken } from '@/lib/jwt';
import type { Kullanici, LoginDto, RegisterDto } from '@/types';

export function AuthProvider({ children }: { children: React.ReactNode }) {
  const [user, setUser] = useState<Kullanici | null>(null);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    const token = sessionStorage.getItem('token');
    const stored = getStoredUser() as Kullanici | null;
    if (token) {
      const fromJwt = userFromToken(token, stored ?? undefined);
      setUser(fromJwt ?? stored);
    }
    setLoading(false);
  }, []);

  const login = useCallback(async (dto: LoginDto) => {
    const res = await authApi.login(dto);
    const token = pickToken(res.data);
    if (!token) {
      throw new Error('Sunucudan token alınamadı.');
    }
    const session = userFromToken(token, { Email: dto.Email });
    if (!session) throw new Error('Token çözümlenemedi.');
    sessionStorage.setItem('token', token);
    sessionStorage.setItem('user', JSON.stringify(session));
    setUser(session);
    return session;
  }, []);

  const register = useCallback(async (dto: RegisterDto) => {
    await authApi.register(dto);
  }, []);

  const logout = useCallback(async () => {
    try {
      await authApi.logout();
    } catch {
      /* token zaten geçersiz olabilir */
    }
    clearAuth();
    setUser(null);
  }, []);

  const hasRole = useCallback(
    (...roles: string[]) => Boolean(user?.Rol && roles.includes(String(user.Rol))),
    [user],
  );

  const value: AuthContextValue = {
    user,
    loading,
    login,
    register,
    logout,
    isAuthenticated: user !== null,
    hasRole,
  };

  return <AuthContext.Provider value={value}>{children}</AuthContext.Provider>;
}

import { useEffect, useState } from 'react';
import { signalR } from '@/services/signalr';

export function useSignalR() {
  const [connectionId, setConnectionId] = useState<string | null>(signalR.connectionId);
  const [aiMesaji, setAiMesaji] = useState<string | null>(null);
  const [connected, setConnected] = useState(false);

  useEffect(() => {
    const offMsg = signalR.onTehis((mesaj) => setAiMesaji(mesaj));
    const offStatus = signalR.onStatus((ok) => {
      setConnected(ok);
      setConnectionId(signalR.connectionId);
    });

    signalR
      .connect()
      .then(() => {
        setConnectionId(signalR.connectionId);
        setConnected(true);
      })
      .catch(() => setConnected(false));

    return () => {
      offMsg();
      offStatus();
    };
  }, []);

  return { connectionId, aiMesaji, setAiMesaji, connected };
}

function isPlainObject(value: unknown): value is Record<string, unknown> {
  return Boolean(value) && typeof value === 'object' && !Array.isArray(value);
}

export function toPascalKeys(value: unknown): unknown {
  if (Array.isArray(value)) return value.map(toPascalKeys);
  if (!isPlainObject(value)) return value;

  const out: Record<string, unknown> = {};
  for (const [key, nested] of Object.entries(value)) {
    const pascal = key.length ? key.charAt(0).toUpperCase() + key.slice(1) : key;
    out[pascal] = toPascalKeys(nested);
  }
  return out;
}

export function apiMessage(data: unknown, fallback: string): string {
  if (typeof data === 'string' && data.trim()) return data;
  if (!isPlainObject(data)) return fallback;
  const mesaj = data.Mesaj ?? data.Hata ?? data.DetayliHata ?? data.message ?? data.title;
  if (typeof mesaj === 'string' && mesaj.trim()) return mesaj;
  if (typeof data.errors === 'object' && data.errors) {
    const first = Object.values(data.errors as Record<string, unknown>)[0];
    if (Array.isArray(first) && typeof first[0] === 'string') return first[0];
  }
  return fallback;
}

export function pickToken(data: unknown): string | null {
  if (!isPlainObject(data)) return null;
  const token = data.Token ?? data.token;
  return typeof token === 'string' && token.length > 0 ? token : null;
}

import type { Kullanici, Role } from '@/types';

const ROLE_CLAIM = 'http://schemas.microsoft.com/ws/2008/06/identity/claims/role';
const NAME_ID_CLAIM = 'http://schemas.xmlsoap.org/ws/2005/05/identity/claims/nameidentifier';

export function decodeJwt(token: string): Record<string, unknown> | null {
  try {
    const payload = token.split('.')[1];
    if (!payload) return null;
    const base64 = payload.replace(/-/g, '+').replace(/_/g, '/');
    const json = decodeURIComponent(
      atob(base64)
        .split('')
        .map((c) => '%' + ('00' + c.charCodeAt(0).toString(16)).slice(-2))
        .join(''),
    );
    return JSON.parse(json);
  } catch {
    return null;
  }
}

export function userFromToken(token: string, extras?: Partial<Kullanici>): Kullanici | null {
  const decoded = decodeJwt(token);
  if (!decoded) return null;

  const idRaw = decoded[NAME_ID_CLAIM] ?? decoded.sub ?? decoded.nameid;
  const rolRaw = decoded[ROLE_CLAIM] ?? decoded.role ?? decoded.Rol;
  const Id = Number(idRaw);
  if (!Id || !rolRaw) return null;

  return {
    Id,
    Rol: String(rolRaw) as Role,
    AdSoyad: extras?.AdSoyad ?? null,
    Email: extras?.Email ?? null,
    Telefon: extras?.Telefon ?? null,
  };
}

export function dashboardPath(rol?: string | null): string {
  switch (rol) {
    case 'SuperAdmin':
    case 'Admin':
      return '/panel';
    case 'Usta':
      return '/usta';
    case 'Veznedar':
      return '/vezne';
    case 'DepoSorumlusu':
      return '/depo';
    case 'Musteri':
      return '/randevularim';
    default:
      return '/';
  }
}

import { useToast } from '@/contexts/ToastContext';
import type { KayitliRandevuTalebiDto } from '@/types';

export function CustomerAppointmentPage() {
  const { connectionId, aiMesaji } = useSignalR();
  const { showSuccess, showError } = useToast();
  const [yeniArac, setYeniArac] = useState(false);
  const [loading, setLoading] = useState(false);
  
  // Tarih ve saati ayrı tutarak daha iyi bir UX sunuyoruz
  const [seciliTarih, setSeciliTarih] = useState('');
  const [seciliSaat, setSeciliSaat] = useState('');

  const [form, setForm] = useState<KayitliRandevuTalebiDto>({
    AracPlaka: '',
    AracMarkaModel: null,
    AracUretimYili: null,
    Tarih: '',
    MusteriNotu: '',
    SignalRConnectionId: '',
  });

  useEffect(() => {
    if (connectionId) setForm((f) => ({ ...f, SignalRConnectionId: connectionId }));
  }, [connectionId]);

  const handleSubmit = async (e: React.FormEvent) => {
    e.preventDefault();
    if (!seciliTarih || !seciliSaat) {
      showError('Eksik Bilgi', 'Lütfen randevu tarihi ve saatini seçiniz.');
      return;
    }

    setLoading(true);
    try {
      // Tarih ve saati birleştirip ISO formatına çeviriyoruz
      const birlesikTarih = new Date(`${seciliTarih}T${seciliSaat}`).toISOString();

        AracPlaka: form.AracPlaka,
        AracMarkaModel: yeniArac ? form.AracMarkaModel : null,
        AracUretimYili: yeniArac ? form.AracUretimYili : null,
        Tarih: birlesikTarih,
        MusteriNotu: form.MusteriNotu,
        SignalRConnectionId: connectionId || form.SignalRConnectionId,
      };
      
      await randevuApi.kayitliMusteri(payload);
      showSuccess('Randevu Talebiniz Alındı', 'Sistem kayıtlarınızı işliyor.');
    } catch (error: unknown) {
      showError('İşlem Başarısız', error instanceof Error ? error.message : 'Tarih seçiminizi kontrol edip tekrar deneyin.');
    } finally {
      setLoading(false);
    }

          <div className="flex-1">
            <h3 className="text-sm font-semibold text-accent-900 mb-1">Yapay Zeka Analizi</h3>
            <p className="text-sm text-accent-800 leading-relaxed">{aiMesaji}</p>
          </div>
        </div>
      )}

      <form onSubmit={handleSubmit} className="card p-6 sm:p-8 space-y-8">
        
        {/* Araç Bilgileri */}
        <div>
          <label className="label-text">Araç Plakası</label>
          <div className="relative mt-1">
            <Car className="absolute left-3 top-1/2 -translate-y-1/2 h-5 w-5 text-ink-400" />
            <input
              name="AracPlaka"
              className="input-field pl-10 uppercase font-medium tracking-wide"
              value={form.AracPlaka}

              required
            />
          </div>

  const [loading, setLoading] = useState(true);

    setLoading(true);

export const api = axios.create({

  headers: { 'Content-Type': 'application/json' },
  transformRequest: [
    (data, headers) => {
      if (headers) headers['Content-Type'] = 'application/json';
      if (data === undefined || data === null) return data;
      if (typeof data === 'string') return data;
      // DTO property adlarını (Sifre, Email, AdSoyad...) birebir PascalCase gönder.
      return JSON.stringify(data);
    },
  ],
});

api.interceptors.request.use((config) => {
  const token = sessionStorage.getItem('token');
  if (token && config.headers) {
    config.headers.Authorization = `Bearer ${token}`;
  }
  return config;
});

api.interceptors.response.use(
  (response) => {
    response.data = toPascalKeys(response.data);
    return response;
  },
  (error) => {
    const data = error.response?.data;
    const normalized = toPascalKeys(data);
    error.message = apiMessage(normalized, error.message || 'İstek başarısız oldu.');
    if (error.response) error.response.data = normalized;
    if (error.response?.status === 401) {
      sessionStorage.removeItem('token');
      sessionStorage.removeItem('user');
    }
    return Promise.reject(error);
  },
);

export function getStoredUser() {
  const raw = sessionStorage.getItem('user');
  try {
    return raw ? JSON.parse(raw) : null;
  } catch {
    return null;
  }
}

export function clearAuth() {
  sessionStorage.removeItem('token');
  sessionStorage.removeItem('user');
}

import { HubConnection, HubConnectionBuilder, LogLevel } from '@microsoft/signalr';
import { API_BASE } from './api';

type MessageHandler = (mesaj: string) => void;
type StatusHandler = (connected: boolean) => void;

class SignalRService {
  private connection: HubConnection | null = null;
  private messageHandlers = new Set<MessageHandler>();
  private statusHandlers = new Set<StatusHandler>();
  connectionId: string | null = null;

  onTehis(handler: MessageHandler): () => void {
    this.messageHandlers.add(handler);
    return () => this.messageHandlers.delete(handler);
  }

  onStatus(handler: StatusHandler): () => void {
    this.statusHandlers.add(handler);
    return () => this.statusHandlers.delete(handler);
  }

  async connect(): Promise<void> {
    if (this.connection?.state === 'Connected') return;

    this.connection = new HubConnectionBuilder()
      .withUrl(`${API_BASE}/notificationHub`)
      .withAutomaticReconnect()
      .configureLogging(LogLevel.Warning)
      .build();

    this.connection.on('ReceiveTehis', (mesaj: string) => {
      this.messageHandlers.forEach((h) => h(mesaj));
    });

    this.connection.onreconnected(() => {
      this.connectionId = this.connection?.connectionId ?? null;
      this.statusHandlers.forEach((h) => h(true));
    });

    this.connection.onclose(() => {
      this.connectionId = null;
      this.statusHandlers.forEach((h) => h(false));
    });

    await this.connection.start();
    this.connectionId = this.connection.connectionId ?? null;
    this.statusHandlers.forEach((h) => h(true));
  }

  async disconnect(): Promise<void> {
    if (!this.connection) return;
    await this.connection.stop();
    this.connection = null;
    this.connectionId = null;
  }
}

export const signalR = new SignalRService();
